#pragma once
/*
   scale — общий заголовок.

   Copyright (c) 2004, by Zhuo Meng (zhuo@thunder.cwru.edu).
   Distributed under GNU GPL v2 or later.

   Здесь собраны:
     - типы ошибок и их строковые представления;
     - структуры TargetSpec / FrameInfo / Args;
     - абстрактный интерфейс Codec (MP2 / AC-3 / DTS Core);
     - фабрики конкретных кодеков;
     - детектор формата по головному буферу;
     - публичные точки входа: parse_args(), parse_target(),
       print_time(), scale_stream().

   Заголовок не содержит тяжёлых зависимостей (только <cstddef>,
   <cstdint>, <expected>, <optional>, <span>, <string>, <string_view>
   плюс print.hpp) — это позволяет использовать его и в main-обёртке,
   и в unit-тестах без лишних include.
*/

#include "scale/print.hpp"

#include <cstddef>
#include <cstdint>
#include <expected>
#include <limits>
#include <optional>
#include <span>
#include <string>
#include <string_view>

namespace scale {

// Версия утилиты. Синхронизируется с project(... VERSION ...)
// в CMakeLists.txt.
inline constexpr std::string_view kVersion = "2.0.1";

// ---------------------------------------------------------------------------
// Internal helpers (not part of public API; may change without notice).
// ---------------------------------------------------------------------------

namespace detail {

// Big-endian 32-bit read from the first 4 bytes of span.
// Caller must ensure b.size() >= 4.
[[nodiscard]] constexpr std::uint32_t
read_be32(std::span<const std::uint8_t> b) noexcept {
    return (static_cast<std::uint32_t>(b[0]) << 24)
         | (static_cast<std::uint32_t>(b[1]) << 16)
         | (static_cast<std::uint32_t>(b[2]) << 8)
         |  static_cast<std::uint32_t>(b[3]);
}


// Sentinel returned by Codec::frame_span() when more data
// is needed (kept alongside read_be32; single source of truth).
inline constexpr std::size_t kNone =
    std::numeric_limits<std::size_t>::max();
} // namespace detail

// ---------------------------------------------------------------------------
// Ошибки
// ---------------------------------------------------------------------------

enum class Error : std::uint8_t {
    Help,          // запрошена справка (-h / --help) — не является ошибкой
    Version,       // запрошена версия (-V / --version) — не является ошибкой
    InvalidArgs,   // некорректные аргументы командной строки
    FileOpen,      // не удалось открыть входной/выходной файл
    FileRead,      // ошибка чтения
    FileWrite,     // ошибка записи в выходной файл
    FileSeek,      // ошибка позиционирования
    BadFormat,     // поток не распознан как MP2 / AC-3 / DTS
    TooShort,      // файл слишком мал для анализа
};

[[nodiscard]] std::string_view to_string(Error e) noexcept;

// ---------------------------------------------------------------------------
// Целевая длительность (-t). Поддерживает три формы:
//   - секунды:       "20.5"   "20.5s"
//   - кадры:         "500f"
//   - время:         "mm:ss"  "hh:mm:ss"  (можно с дробной частью секунд)
// ---------------------------------------------------------------------------

struct TargetSpec {
    enum class Kind : std::uint8_t {
        Seconds,   // value — секунды
        Frames,    // value — число кадров
    };

    Kind   kind  = Kind::Seconds;
    double value = 0.0;

    [[nodiscard]] bool active() const noexcept { return value > 0.0; }
};

// ---------------------------------------------------------------------------
// Информация о кадре
// ---------------------------------------------------------------------------

struct FrameInfo {
    std::size_t framebytes   = 0;    // размер ядра кадра (без DTS-HD extension)
    double      framesec     = 0.0;  // длительность одного кадра, сек
    double      freq_khz     = 0.0;  // частота дискретизации, кГц
    std::size_t bitrate_kbps = 0;    // битрейт, кбит/с
};

// ---------------------------------------------------------------------------
// Аргументы командной строки
// ---------------------------------------------------------------------------

struct Args {
    TargetSpec  target;    // не active() => только отчёт, без масштабирования
    std::string outfile;   // пусто => без записи выходного файла
    std::string infile;
};

// ---------------------------------------------------------------------------
// Интерфейс кодека
//
//   Все методы — noexcept и bounds-safe: любая проверка размера буфера
//   выполняется внутри parse_header / frame_span, ошибки возвращаются
//   через std::expected. Движок (common.cpp) никогда не читает буфер
//   напрямую, а только через эти методы.
// ---------------------------------------------------------------------------

class Codec {
public:
    virtual ~Codec() = default;

    Codec(const Codec&)            = delete;
    Codec& operator=(const Codec&) = delete;

    // Человекочитаемые имена: "mp2" / "ac3" / "dts" и "MP2" / "AC-3" / "DTS".
    // name() — для отладочных сообщений, long_name() — для сводки scale_stream().
    [[nodiscard]] virtual std::string_view name()      const noexcept = 0;
    [[nodiscard]] virtual std::string_view long_name() const noexcept = 0;

    // Проверить синхрослово в начале span (нужно 2..4 байта для
    // MP2/AC-3 и 4 байта для DTS Core / DTS-HD).
    [[nodiscard]] virtual bool
        has_sync(std::span<const std::uint8_t> buf) const noexcept = 0;

    // Разобрать заголовок кадра. Требует минимум 4..10 байт (зависит
    // от формата). Возвращает FrameInfo либо BadFormat / TooShort.
    [[nodiscard]] virtual std::expected<FrameInfo, Error>
        parse_header(std::span<const std::uint8_t> buf) const noexcept = 0;

    // Признак того, что после ядра кадра может идти extension
    // (DTS-HD MA/HRA). Движок использует его, чтобы не требовать
    // следующий sync сразу за ядром для таких форматов.
    //
    // Extension point: codecs that carry extension data after the
    // core frame (DTS-HD XLL/X96, future TrueHD, Opus-in-Ogg)
    // override this to skip the immediate-next-sync check in
    // find_frame() / for_each_frame().
    [[nodiscard]] virtual bool has_extension() const noexcept {
        return false;
    }

    // Расстояние от текущего sync до следующего (>= fi.framebytes).
    //   * SIZE_MAX → данных недостаточно, движок должен дозаполнить буфер;
    //   * базовое поведение — вернуть fi.framebytes (MP2, AC-3);
    //   * DTS Core переопределяет: сканирует следующий sync, что
    //     автоматически отбрасывает XLL/HRA-расширение DTS-HD и даёт
    //     на выходе чистый DTS Core.
    [[nodiscard]] virtual std::size_t
        frame_span(std::span<const std::uint8_t> buf,
                   const FrameInfo& fi) const noexcept {
        static_cast<void>(buf);
        return fi.framebytes;
    }

    // Напечатать человекочитаемую сводку о потоке (использует первый
    // кадр и уже распарсенный FrameInfo).
    virtual void print_info(std::span<const std::uint8_t> first_frame,
                            const FrameInfo& fi) const = 0;

protected:
    Codec() = default;
};

// Фабрики конкретных кодеков (thread-safe: статические локальные).
[[nodiscard]] const Codec& mp2_codec();
[[nodiscard]] const Codec& ac3_codec();
[[nodiscard]] const Codec& dts_codec();

// ---------------------------------------------------------------------------
// Детектор формата
// ---------------------------------------------------------------------------

struct Detected {
    const Codec* codec  = nullptr;
    std::size_t  offset = 0;   // смещение первого валидного кадра
    FrameInfo    fi{};
};

// Пробует DTS → AC-3 → MP2. Для DTS-HD MA входной поток содержит
// DTS Core в начале кадра, поэтому детектируется тем же синхрословом.
//
// total_size — полный размер входного файла. 0 означает head.size()
// (head содержит весь файл). Передавайте фактический размер, когда
// head — усечённый префикс: это позволяет find_frame() отличать
// «кадр в конце файла» от «кадр в конце head» и распознавать
// одиночный кадр с хвостовым мусором (relax-проход).
[[nodiscard]] std::optional<Detected>
    detect_codec(std::span<const std::uint8_t> head,
                 std::size_t total_size = 0) noexcept;

// ---------------------------------------------------------------------------
// Публичные точки входа
// ---------------------------------------------------------------------------

// Разобрать argc/argv. Возвращает Args либо:
//   * Error::Help        — при -h / --help (usage напечатан в stdout);
//   * Error::Version     — при -V / --version (версия напечатана в stdout);
//   * Error::InvalidArgs — при некорректных опциях или отсутствии
//                          входного файла (usage напечатан в stderr).
// Обработка ошибки остаётся за вызывающим (main).
[[nodiscard]] std::expected<Args, Error>
    parse_args(int argc, char** argv, std::string_view prg);

// Разобрать значение -t. Возвращает nullopt при невалидной строке.
//   "20.5"      → 20.5 s
//   "20.5s"     → 20.5 s
//   "500f"      → 500 frames
//   "00:20:30"  → 20 min 30 sec
//   "01:30"     → 1 min 30 sec
[[nodiscard]] std::optional<TargetSpec>
    parse_target(std::string_view s) noexcept;

// Печать времени в формате HH:MM:SS.mmm.
void print_time(double t);

// Полный цикл: детект формата, анализ файла, построение плана и (если
// задан outfile) масштабирование потока. Многопоточно: параллельный
// head-scan + конвейер reader/writer. Внутри сам выбирает Codec через
// detect_codec(), поэтому принимает только Args.
[[nodiscard]] std::expected<void, Error>
    scale_stream(const Args& args);

} // namespace scale

#pragma once
/*
   scale/print.hpp — единый shim для std::print / std::println.

   GCC 13 и Clang 17 (заявленный минимум проекта) не имеют <print> в
   libstdc++/libc++. Чтобы не дублировать fallback в каждом .cpp,
   он вынесен в этот заголовок. Если __cpp_lib_print доступен —
   используем стандартный std::print; иначе — тонкую обёртку над
   std::vformat + std::fwrite.

   Использование:
       #include "scale/print.hpp"
       ...
       scale::print("value = {}\n", v);
       scale::print(stderr, "error: {}\n", msg);
       scale::println("line with newline");

   Все функции принимают std::format-style формат и аргументы.
   scale::print(FILE*, fmt, args...)   — печать в произвольный FILE*
   scale::print(fmt, args...)          — печать в stdout
   scale::println(FILE*, fmt, args...) — то же + '\n'
   scale::println(fmt, args...)        — то же в stdout + '\n'
*/

#include <cstdio>
#include <format>
#include <string_view>
#include <utility>
#include <version>

#if defined(__cpp_lib_print) && __cpp_lib_print >= 202207L

// ---------------------------------------------------------------------------
// Стандартный <print> доступен (GCC 14+, Clang 18+ с libstdc++ 14+,
// MSVC 19.37+ с /std:c++latest или /std:c++23preview).
// ---------------------------------------------------------------------------
#  include <print>

namespace scale {
    using std::print;
    using std::println;
} // namespace scale

#else

// ---------------------------------------------------------------------------
// Fallback: <print> отсутствует. Реализуем через std::vformat + fwrite.
// ---------------------------------------------------------------------------
namespace scale {

// --- print(FILE*, fmt, args...) --------------------------------------------
template <class... Args>
inline void print(std::FILE* f, std::string_view fmt, Args&&... args) {
    auto s = std::vformat(fmt, std::make_format_args(args...));
    std::fwrite(s.data(), 1, s.size(), f);
}

// --- print(fmt, args...) — в stdout ----------------------------------------
template <class... Args>
inline void print(std::string_view fmt, Args&&... args) {
    print(stdout, fmt, std::forward<Args>(args)...);
}

// --- println(FILE*, fmt, args...) ------------------------------------------
// Mirrors std::println(FILE*, ...) API. Used internally to
// avoid duplicating `print(...) + fputc`.
template <class... Args>
inline void println(std::FILE* f, std::string_view fmt, Args&&... args) {
    print(f, fmt, std::forward<Args>(args)...);
    std::fputc('\n', f);
}

// --- println(fmt, args...) — в stdout --------------------------------------
template <class... Args>
inline void println(std::string_view fmt, Args&&... args) {
    println(stdout, fmt, std::forward<Args>(args)...);
}

} // namespace scale

#endif // __cpp_lib_print

#pragma code_page(65001)
// =============================================================================
//  scale.rc.in — шаблон ресурсного файла для Windows-сборки scale.
//
//  CMake подставляет @PROJECT_VERSION_*@ при конфигурации и генерирует
//  scale.rc в бинарном каталоге. Пути к манифесту и иконке берутся из
//  @CMAKE_CURRENT_SOURCE_DIR@ (forward slashes понимают и rc.exe, и windres).
//
//  Содержит три ресурса:
//    * иконку приложения (RT_GROUP_ICON, IDI_ICON1);
//    * манифест приложения (RT_MANIFEST, ID 1);
//    * информацию о версии (VERSIONINFO).
// =============================================================================

#include <windows.h>

// -----------------------------------------------------------------------------
// Иконка приложения.
//
// IDI_ICON1 — стандартный идентификатор (обычно 1). Windows Explorer
// выбирает иконку с наименьшим ID среди RT_GROUP_ICON, поэтому 1 —

/*
   ac3codec.cpp — реализация scale::Codec для AC-3 (Dolby Digital).

   Copyright (c) 2004, by Zhuo Meng (zhuo@thunder.cwru.edu).
   Distributed under GNU GPL v2 or later.

   Класс Ac3Codec инкапсулирует всё, что относится к формату AC-3:
     - синхрослово 0x0B77 и его проверку;
     - разбор заголовка кадра (fscod / frmsizecod / acmod / bsmod / lfeon);
     - вычисление размера кадра в байтах и его длительности;
     - печать человекочитаемой сводки о потоке (версия, тип сервиса,
       конфигурация каналов, LFE).

   Таблицы соответствуют ATSC A/52 (AC-3), §5.3 "Sync Frame".
   Размеры кадра заданы отдельно для 48, 44.1 и 32 кГц.
*/

// ---------------------------------------------------------------------------
// Таблицы AC-3 (ATSC A/52)
// ---------------------------------------------------------------------------

// frmsizecod (6 бит) -> битрейт, кбит/с. Индексы 0..37 валидны,
// 38..63 зарезервированы. Значения идут парами: для каждой пары
// (2k, 2k+1) битрейт одинаков, отличается только выбор размера кадра.
constexpr std::array<std::size_t, 38> kBitrates = {
    32,  32,  40,  40,  48,  48,  56,  56,
    64,  64,  80,  80,  96,  96, 112, 112,
   128, 128, 160, 160, 192, 192, 224, 224,
   256, 256, 320, 320, 384, 384, 448, 448,
   512, 512, 576, 576, 640, 640
};

// fscod (2 бита) -> частота дискретизации, кГц.
//   00 = 48 kHz, 01 = 44.1 kHz, 10 = 32 kHz, 11 = reserved.
constexpr std::array<double, 4> kFreqs = { 48.0, 44.1, 32.0, 0.0 };

// Число сэмплов на кадр AC-3 (фиксировано стандартом).
constexpr std::size_t kSamplesPerFrame = 1536;

// Минимально допустимый размер кадра в байтах:
//   2 sync + 2 crc1 + 2 bSI + 2 bsi — минимальный валидный кадр.
constexpr std::size_t kMinFrameBytes = 8;

// Максимальный размер кадра AC-3 = 3840 байт (640 кбит/с при 32 кГц,
// 1536 сэмплов). Верхняя защита от мусорных frmsizecod.
constexpr std::size_t kMaxFrameBytes = 3840;

// Размер кадра в 16-битных словах, индексируется [frmsizecod][fscod].
// Колонки: 48 кГц, 44.1 кГц, 32 кГц.
//
// Источник: ATSC A/52 §5.3 (совпадает с ff_ac3_frame_size_tab из FFmpeg).
constexpr std::array<std::array<std::size_t, 3>, 38> kFrameSizeWords = {{
    {  64,  69,  96 }, {  64,  70,  96 },
    {  80,  87, 120 }, {  80,  88, 120 },
    {  96, 104, 144 }, {  96, 105, 144 },
    { 112, 121, 168 }, { 112, 122, 168 },
    { 128, 139, 192 }, { 128, 140, 192 },
    { 160, 174, 240 }, { 160, 175, 240 },
    { 192, 208, 288 }, { 192, 209, 288 },
    { 224, 243, 336 }, { 224, 244, 336 },
    { 256, 278, 384 }, { 256, 279, 384 },
    { 320, 348, 480 }, { 320, 349, 480 },
    { 384, 417, 576 }, { 384, 418, 576 },
    { 448, 487, 672 }, { 448, 488, 672 },
    { 512, 557, 768 }, { 512, 558, 768 },
    { 640, 696, 960 }, { 640, 697, 960 },
    { 768, 835,1152 }, { 768, 836,1152 },
    { 896, 975,1344 }, { 896, 976,1344 },
    {1024,1114,1536 }, {1024,1115,1536 },
    {1152,1253,1728 }, {1152,1254,1728 },
    {1280,1393,1920 }, {1280,1394,1920 }
}};

// ---------------------------------------------------------------------------
// Реализация scale::Codec для AC-3
// ---------------------------------------------------------------------------

class Ac3Codec final : public scale::Codec {
public:
    // -----------------------------------------------------------------------
    // Метаданные
    // -----------------------------------------------------------------------
    [[nodiscard]] std::string_view name()      const noexcept override { return "ac3"; }
    [[nodiscard]] std::string_view long_name() const noexcept override { return "AC-3"; }

    // -----------------------------------------------------------------------
    // Синхрослово AC-3: 0x0B77 (16 бит, big-endian).
    // -----------------------------------------------------------------------
    [[nodiscard]] bool
    has_sync(std::span<const std::uint8_t> b) const noexcept override {
        return b.size() >= 2
            && b[0] == 0x0B
            && b[1] == 0x77;
    }

    // -----------------------------------------------------------------------
    // Разбор заголовка кадра.
    //
    // Требуется минимум 8 байт:
    //   b[0..1]  sync 0x0B77
    //   b[2..3]  crc1
    //   b[4]     fscod (2) | frmsizecod (6)
    //   b[5]     bsid  (5) | bsmod   (3)
    //   b[6]     acmod (3) | cmixlev/surmixlev/dsurmod/lfeon (остальное)
    //   b[7]     ...продолжение bsi...
    // -----------------------------------------------------------------------
    [[nodiscard]] std::expected<scale::FrameInfo, scale::Error>
    parse_header(std::span<const std::uint8_t> b) const noexcept override {
        using scale::Error;
        using scale::FrameInfo;

        // Синхрослово обязательно + отсекаем E-AC-3 по bsid (16).
        if (b.size() < 8 || !has_sync(b))
            return std::unexpected(Error::BadFormat);

        // bsid 0..10 — AC-3, 16 — E-AC-3, остальные — резерв.
        // E-AC-3 имеет иную структуру кадра; здесь не поддержан.
        if (((b[5] & 0xF8u) >> 3) > 10)
            return std::unexpected(Error::BadFormat);

        // Байт 4: fscod в старших 2 битах, frmsizecod в младших 6.
        const std::uint8_t b4 = b[4];

        const std::size_t fscod      =
            static_cast<std::size_t>((b4 & 0xC0u) >> 6);
        const std::size_t frmsizecod =
            static_cast<std::size_t>(b4 & 0x3Fu);

        // Валидация индексов — иначе OOB и деление на ноль.
        if (fscod >= 3 || kFreqs[fscod] == 0.0)
            return std::unexpected(Error::BadFormat);
        if (frmsizecod >= kBitrates.size())
            return std::unexpected(Error::BadFormat);

        // Размер кадра берём напрямую из таблицы ATSC A/52.
        const std::size_t words      = kFrameSizeWords[frmsizecod][fscod];
        const std::size_t framebytes = words * 2;

        if (framebytes < kMinFrameBytes || framebytes > kMaxFrameBytes)
            return std::unexpected(Error::BadFormat);

        FrameInfo fi{};
        fi.framebytes   = framebytes;
        fi.framesec     = static_cast<double>(kSamplesPerFrame)
                        / (kFreqs[fscod] * 1000.0);
        fi.freq_khz     = kFreqs[fscod];
        fi.bitrate_kbps = kBitrates[frmsizecod];
        return fi;
    }

    // -----------------------------------------------------------------------
    // frame_span() для AC-3 не переопределяем.
    //
    // Кадры AC-3 фиксированного размера; frmsizecod кодирует уже финальный
    // размер, отдельного padding-бита нет. Следующий sync находится ровно
    // на fi.framebytes от текущего — базовая реализация Codec::frame_span()
    // возвращает именно это значение.
    // -----------------------------------------------------------------------

    // -----------------------------------------------------------------------
    // Человекочитаемая сводка о потоке.
    // -----------------------------------------------------------------------
    void print_info(std::span<const std::uint8_t> b,
                    const scale::FrameInfo& fi) const override {
        if (b.size() < 8) return;

        // Байт 5: bsid (5 бит) + bsmod (3 бита).
        // Байт 6: acmod (3 бита) + остальное.
        const std::uint8_t b5 = b[5];
        const std::uint8_t b6 = b[6];

        const std::size_t ac3v  =
            static_cast<std::size_t>((b5 & 0xF8u) >> 3);
        const std::size_t bsmod =
            static_cast<std::size_t>(b5 & 0x07u);
        const std::size_t acmod =
            static_cast<std::size_t>((b6 & 0xE0u) >> 5);

        // Режимы каналов AC-3 (acmod, 3 бита):
        //   0 = 1+1 (Ch1,Ch2)
        //   1 = 1/0 (C)
        //   2 = 2/0 (L,R)
        //   3 = 3/0 (L,C,R)
        //   4 = 2/1 (L,R,S)
        //   5 = 3/1 (L,C,R,S)
        //   6 = 2/2 (L,R,SL,SR)
        //   7 = 3/2 (L,C,R,SL,SR)
        static constexpr std::array<std::string_view, 8> kChanMode = {
            "1+1", "1/0", "2/0", "3/0", "2/1", "3/1", "2/2", "3/2"
        };
        static constexpr std::array<std::size_t, 8> kNfChans = {
            2, 1, 2, 3, 3, 4, 4, 5
        };

        // Позиция lfeon зависит от acmod:
        //   acmod 1/0  — сдвиг 4;
        //   acmod 1+1  — сдвиг 4;
        //   acmod 2/0  — сдвиг 4;
        //   acmod 3/0  — сдвиг 3 (2 бита cmixlev);
        //   acmod 2/1  — сдвиг 4;
        //   acmod 3/1  — сдвиг 3;
        //   acmod 2/2  — сдвиг 3 (2 бита dsurmod);
        //   acmod 3/2  — сдвиг 2 (cmixlev + surmixlev).
        // Каждое присутствующее 2-битное поле (cmixlev, surmixlev,
        // dsurmod) сдвигает lfeon на 2 бита вправо относительно
        // конца acmod (старшие 3 бита байта 6).
        std::size_t shift = 4;
        if ((acmod & 0x01) && acmod != 0x01) shift -= 2;   // cmixlev
        if ((acmod & 0x04) || acmod == 0x02) shift -= 2;   // surmixlev / dsurmod
        const bool lfe = ((b6 >> shift) & 1u) != 0;

        // bsmod: 0 = complete main, 1 = music/effects, 2 = visually impaired,
        //        3 = hearing impaired, 4 = dialogue, 5 = commentary,
        //        6 = emergency, 7 = voice-over (для acmod > 1).
        // "Main audio service" — bsmod 0 или 1, а также 7 для многоканальных.
        print("Audio type is AC-3, version {}, ", ac3v);
        if (bsmod < 2 || (bsmod == 7 && acmod > 1))
            print("main audio service, ");
        else
            print("associated service, ");

        print("{}{} channel(s) mode {}.\n",
              kNfChans[acmod], lfe ? ".1" : "", kChanMode[acmod]);

        print("Each frame is {} bytes, sampling freq is {} kHz, "
              "and bitrate is {} kbps.\n",
              fi.framebytes, fi.freq_khz, fi.bitrate_kbps);
    }
};

} // namespace

    static const Ac3Codec instance;
    return instance;
}

// ---------------------------------------------------------------------------
// Ограниченная очередь producer/consumer с корректным завершением.
//   push   возвращает false, если очередь закрыта;
//   pop    возвращает nullopt, если очередь пуста И закрыта.
// ---------------------------------------------------------------------------
template <class T>
class BoundedQueue {
public:
    explicit BoundedQueue(std::size_t cap) : cap_(cap) {}
    BoundedQueue(const BoundedQueue&)            = delete;
    BoundedQueue& operator=(const BoundedQueue&) = delete;

    bool push(T v) {
        std::unique_lock lk(m_);
        cv_not_full_.wait(lk, [&] { return q_.size() < cap_ || closed_; });
        if (closed_) return false;
        q_.push(std::move(v));
        cv_not_empty_.notify_one();
        return true;
    }

    std::optional<T> pop() {
        std::unique_lock lk(m_);
        cv_not_empty_.wait(lk, [&] { return !q_.empty() || closed_; });
        if (q_.empty()) return std::nullopt;
        T v = std::move(q_.front());
        q_.pop();
        cv_not_full_.notify_one();
        return v;
    }

    void close() {
        std::lock_guard lk(m_);
        closed_ = true;
        cv_not_empty_.notify_all();
        cv_not_full_.notify_all();
    }

private:
    std::size_t             cap_;
    std::queue<T>           q_;
    std::mutex              m_;
    std::condition_variable cv_not_empty_, cv_not_full_;
    bool                    closed_ = false;
};

// Кадр, передаваемый по конвейеру. Хранит только ядро (без DTS-HD
// extension) — движок всегда пишет DTS Core.
struct FrameBlock {
    std::vector<std::uint8_t> bytes;
    double                    framesec = 0.0;
};

// Действие над кадром в плане масштабирования.
enum class Action : std::uint8_t { Copy, Drop, Dup };

// Распределить ровно `count` действий `action` по `nframes` позициям
// Bresenham-подобным алгоритмом (равномерно, без скоплений).
// Используется как для дублирования, так и для удаления кадров.
//
// При count >= nframes каждая позиция получит действие (все Dup
// либо все Drop) — это используется вызывающим кодом для обработки
// недостижимой цели.
void distribute_actions(std::vector<Action>& plan,
                        std::size_t nframes,
                        long count,
                        Action action) {
    if (count <= 0 || nframes == 0) return;
    const long total = static_cast<long>(nframes);
    long acc = 0;
    for (std::size_t i = 0; i < nframes; ++i) {
        acc += count;
        if (acc >= total) {
            acc -= total;
            plan[i] = action;
        }
    }
}

// ---------------------------------------------------------------------------
// Разбор double с полной проверкой.
// ---------------------------------------------------------------------------
bool parse_double(std::string_view s, double& out) {
    if (s.empty()) return false;
#if defined(__cpp_lib_to_chars) && __cpp_lib_to_chars >= 201611L
    // Локаль-независимый парсер с плавающей точкой.
    const char* first = s.data();
    const char* last  = s.data() + s.size();
    auto [ptr, ec] = std::from_chars(first, last, out);
    if (ec != std::errc{} || ptr != last) return false;
#else
    try {
        std::size_t pos = 0;
        out = std::stod(std::string(s), &pos);
        if (pos != s.size()) return false;
    } catch (...) {
        return false;
    }
#endif
    return std::isfinite(out) && out >= 0.0;
}

// ---------------------------------------------------------------------------
// Bounds-safe поиск первого полного кадра в буфере.
//
//   total_size           — полный размер входного файла (0 = buf.size());
//   allow_trailing_junk  — разрешить одиночный кадр в начале потока,
//                          за которым нет валидного sync (хвостовой
//                          мусор). Применяется только на relax-проходе
//                          detect_codec(), чтобы не перекрывать строгое
//                          совпадение другого кодека.
// ---------------------------------------------------------------------------
std::optional<std::pair<std::size_t, FrameInfo>>
find_frame(std::span<const std::uint8_t> buf, const Codec& c,
           std::size_t total_size = 0,
           bool allow_trailing_junk = false) {
    const std::size_t n = buf.size();
    if (n < 10) return std::nullopt;
    if (total_size == 0) total_size = n;

    for (std::size_t i = 0; i + 10 <= n; ++i) {
        auto view = buf.subspan(i);
        if (!c.has_sync(view)) continue;

        auto fi = c.parse_header(view);
        if (!fi) continue;

        const std::size_t core = fi->framebytes;
        if (core > kMaxFrame) continue;             // защитный верхний предел
        if (i + core > total_size) continue;        // кадр не влезает в файл

        // Для MP2 / AC-3 сразу за ядром должен идти следующий sync —
        // это отсеивает случайные совпадения 0xFFF / 0x0B77 в мусоре.
        // Для DTS после ядра может быть XLL/HRA extension, поэтому
        // проверку не делаем (детектор всё равно попробует DTS первым).
        if (!c.has_extension()) {
            // Кадр заканчивается ровно в конце файла — одиночный кадр.
            if (i + core == total_size) return std::pair{i, *fi};
            // Иначе проверяем следующий sync, если он влезает в буфер.
            if (i + core <= n) {
                auto next = buf.subspan(i + core);
                if (c.has_sync(next) && c.parse_header(next))
                    return std::pair{i, *fi};
            }
            // Хвостовой мусор после одиночного кадра: принимаем,
            // только если кадр идёт с самого начала и это relax-проход.
            // Послабление позволяет детектировать поток из одного кадра
            // с «прицепом», не открывая дорогу ложным срабатываниям
            // sync-слов в середине мусорного файла.
            if (allow_trailing_junk && i == 0)
                return std::pair{i, *fi};
            continue;
        }

        return std::pair{i, *fi};
    }
    return std::nullopt;
}

// ---------------------------------------------------------------------------
// Проверяет, начинается ли DTS-HD extension (XLL/X96) по указанному
// смещению в буфере. Возвращает false, если данных недостаточно.
[[nodiscard]] bool has_dts_hd_extension(std::span<const std::uint8_t> buf,
                                        std::size_t offset) noexcept {
    if (offset > buf.size() || buf.size() - offset < 4) return false;
    const std::uint32_t sig = detail::read_be32(buf.subspan(offset, 4));
    // DTS-HD MA: 0x41A29547; DTS-HD HRA: 0x1D95F262.
    return sig == 0x41A29547u || sig == 0x1D95F262u;
}

// ---------------------------------------------------------------------------
// Сколько байт занимает последний кадр (когда frame_span() == kNone).
//
//   frame_buf  — данные от sync текущего кадра до конца доступного буфера;
//   core       — размер ядра (fi.framebytes);
//   byte_limit — максимум байт, которые можно взять. Для конвейера это
//                remaining в файле; для tail-сканирования — весь остаток
//                буфера (frame_buf.size()).
//
// Если сразу за ядром идёт сигнатура DTS-HD extension (XLL/X96) — берём
// весь доступный остаток (ограниченный byte_limit). Иначе — только ядро.
// has_dts_hd_extension() сам отсекает случай core+4 > buf.size(),
// поэтому чтения за границей буфера здесь нет.
// ---------------------------------------------------------------------------
[[nodiscard]] std::size_t
last_frame_extent(std::span<const std::uint8_t> frame_buf,
                  std::size_t core,
                  std::size_t byte_limit) noexcept {
    if (has_dts_hd_extension(frame_buf, core))
        return std::min(byte_limit, frame_buf.size());
    return core;
}

// ---------------------------------------------------------------------------
// Размер хвостового мусора в окне tail: сколько байт в конце не
// принадлежат последнему полному кадру.
// ---------------------------------------------------------------------------
std::size_t tail_junk_size(std::span<const std::uint8_t> tail, const Codec& c) {
    std::size_t last_end = 0;

    for (std::size_t i = 0; i + 10 <= tail.size(); ++i) {
        if (!c.has_sync(tail.subspan(i))) continue;
        auto fi = c.parse_header(tail.subspan(i));
        if (!fi) continue;
        if (i + fi->framebytes > tail.size()) continue;

        const std::size_t span = c.frame_span(tail.subspan(i), *fi);

        std::size_t end;
        if (span == kNone || i + span > tail.size()) {
            // frame_span не нашёл следующий sync — это последний кадр.
            // last_frame_extent() решает, брать только ядро или
            // ядро+extension (DTS-HD MA/HRA).
            const auto frame_view = tail.subspan(i);
            end = i + last_frame_extent(frame_view, fi->framebytes,
                                        frame_view.size());
        } else {
            end = i + span;
        }
        if (end > last_end) last_end = end;
    }

    // Если в окне нет ни одного кадра — весь хвост является мусором.
    if (last_end == 0) return tail.size();
    return tail.size() - last_end;
}

// ---------------------------------------------------------------------------
// Результат обхода потока кадр за кадром.
// ---------------------------------------------------------------------------
struct ScanResult {
    std::size_t nframes   = 0;
    double      total_sec = 0.0;
    bool        ok        = false;
};

// ---------------------------------------------------------------------------
// Статус обхода:
//   Ok    — поток полностью обработан (остаток len равен нулю);
//   Eof   — достигнут конец файла либо callback попросил остановиться;
//   Error — ошибка ввода-вывода или невалидный кадр.
// ---------------------------------------------------------------------------
enum class ReadStatus { Ok, Eof, Error };

// ---------------------------------------------------------------------------
// Универсальный обход потока кадр за кадром.
//
// Один и тот же цикл используется и для подсчёта кадров (scan_frames),
// и для конвейера reader/writer в scale_stream(), что исключает
// дублирование логики обработки kNone / переполнения буфера.
//
// Callback получает span ровно на core байт (само ядро кадра без
// DTS-HD extension) и FrameInfo. Возврат true — продолжать, false —
// прекратить обход (эквивалентно ReadStatus::Eof).
// ---------------------------------------------------------------------------
template <class Fn>
ReadStatus for_each_frame(std::FILE* fin, std::size_t start, std::size_t len,
                          const Codec& codec, Fn&& cb,
                          std::stop_token st = {}) {
    if (SCALE_FSEEK(fin, static_cast<std::int64_t>(start), SEEK_SET) != 0)
        return ReadStatus::Error;

    std::vector<std::uint8_t> buf(kBufSize);
    std::size_t have = 0, pos = 0, remaining = len;
    bool eof_reached = false;

    auto refill = [&]() -> ReadStatus {
        std::memmove(buf.data(), buf.data() + pos, have - pos);
        have -= pos;
        pos   = 0;
        const std::size_t requested = buf.size() - have;
        const std::size_t got = std::fread(buf.data() + have, 1,
                                           requested, fin);
        if (std::ferror(fin)) return ReadStatus::Error;
        if (got < requested) eof_reached = true;
        have += got;
        return ReadStatus::Ok;
    };

    while (remaining > 0) {
        if (st.stop_requested()) return ReadStatus::Eof;

        if (pos == have) {
            have = std::fread(buf.data(), 1, buf.size(), fin);
            pos  = 0;
            if (std::ferror(fin)) return ReadStatus::Error;
            if (have < buf.size()) eof_reached = true;
            if (have == 0) return ReadStatus::Eof;
        }

        // Ensure at least kMaxHeaderBytes (10 for DTS) bytes are
        // available before header parse. Otherwise a frame boundary
        // near the buffer end causes a spurious BadFormat error.
        constexpr std::size_t kMaxHeaderBytes = 10;
        while (!eof_reached && (have - pos) < kMaxHeaderBytes) {
            const ReadStatus rs = refill();
            if (rs == ReadStatus::Error) return rs;
        }

        auto sp = std::span<const std::uint8_t>{buf.data() + pos, have - pos};
        auto fi = codec.parse_header(sp);
        if (!fi) return ReadStatus::Error;

        const std::size_t core = fi->framebytes;
        std::size_t span = codec.frame_span(sp, *fi);

        if (span == kNone || pos + span > have) {
            // Пытаемся дочитать данные, только если файл ещё не
            // закончился. После EOF это как раз тот случай, когда
            // последний кадр DTS остаётся в буфере без sync.
            if (!eof_reached && (pos > 0 || have < buf.size())) {
                const ReadStatus rs = refill();
                if (rs == ReadStatus::Error) return rs;
                continue;
            }
            if (span == kNone) {
                if (!eof_reached) {
                    // Buffer full but frame doesn't fit: exceeds kBufSize.
                    // Refuse rather than silently truncating data.
                    return ReadStatus::Error;
                }
                // Последний кадр в файле. last_frame_extent()
                // решает, брать только ядро или ядро+extension
                // (DTS-HD MA/HRA).
                const auto frame_view = std::span<const std::uint8_t>{
                    buf.data() + pos, have - pos};
                span = last_frame_extent(frame_view, core, remaining);
            } else if (pos + span > have) {
                return ReadStatus::Error;
            }
        }

        if (core > span || span > remaining) return ReadStatus::Error;
        if (pos + core > have) return ReadStatus::Error;

        auto frame = std::span<const std::uint8_t>{buf.data() + pos, core};
        if (!cb(frame, *fi)) return ReadStatus::Eof;

        pos       += span;
        remaining -= span;
    }
    return ReadStatus::Ok;
}

// ---------------------------------------------------------------------------
// Тонкая обёртка над for_each_frame(): подсчёт кадров и их суммарной
// длительности. Используется перед построением плана dup/drop.
// ---------------------------------------------------------------------------
ScanResult scan_frames(std::FILE* fin, std::size_t start, std::size_t len,
                       const Codec& codec) {
    ScanResult r;
    const ReadStatus st = for_each_frame(fin, start, len, codec,
        [&](std::span<const std::uint8_t>, const FrameInfo& fi) {
            ++r.nframes;
            r.total_sec += fi.framesec;
            return true;
        });
    r.ok = (st == ReadStatus::Ok);
    return r;
}

std::string_view to_string(Error e) noexcept {
    switch (e) {
        case Error::Help:        return "help requested";
        case Error::Version:     return "version requested";
        case Error::InvalidArgs: return "invalid arguments";
        case Error::FileOpen:    return "cannot open file";
        case Error::FileRead:    return "read error";
        case Error::FileWrite:   return "write error";
        case Error::FileSeek:    return "seek error";
        case Error::BadFormat:   return "unrecognized audio format";
        case Error::TooShort:    return "file is too short";
    }
    return "unknown error";
}

// ---------------------------------------------------------------------------
// Парсинг -t:
//   "20.5"     → 20.5 s
//   "20.5s"    → 20.5 s
//   "500f"     → 500 frames
//   "01:30"    → 1 min 30 sec
//   "00:20:30" → 20 min 30 sec
// ---------------------------------------------------------------------------
std::optional<TargetSpec> parse_target(std::string_view s) noexcept {
    if (s.empty()) return std::nullopt;

    // --- форма hh:mm:ss или mm:ss ---
    if (s.find(':') != std::string_view::npos) {
        // Фиксированный массив вместо std::vector: parse_target()
        // помечен noexcept, а push_back в vector может бросить
        // bad_alloc → std::terminate(). Максимум 3 компоненты
        // (hh:mm:ss); 4-я — невалидный ввод.
        std::array<std::string_view, 3> parts{};
        std::size_t nparts = 0;
        std::size_t start  = 0;
        while (start <= s.size()) {
            if (nparts >= parts.size()) return std::nullopt;  // 4-я компонента
            auto colon = s.find(':', start);
            auto end   = (colon == std::string_view::npos) ? s.size() : colon;
            parts[nparts++] = s.substr(start, end - start);
            if (colon == std::string_view::npos) break;
            start = colon + 1;
        }
        if (nparts < 2) return std::nullopt;

        static constexpr double kScale[3] = { 3600.0, 60.0, 1.0 };
        const std::size_t off = 3 - nparts;

        double total = 0.0;
        for (std::size_t i = 0; i < nparts; ++i) {
            double v = 0.0;
            if (!parse_double(parts[i], v)) return std::nullopt;
            total += v * kScale[off + i];
        }
        if (total <= 0.0) return std::nullopt;

        TargetSpec t;
        t.kind  = TargetSpec::Kind::Seconds;
        t.value = total;
        return t;
    }

    // --- суффикс 's' / 'S' → секунды ---
    const char last = s.back();
    if (last == 's' || last == 'S') {
        double v = 0.0;
        if (!parse_double(s.substr(0, s.size() - 1), v)) return std::nullopt;
        if (v <= 0.0) return std::nullopt;
        TargetSpec t;
        t.kind  = TargetSpec::Kind::Seconds;
        t.value = v;
        return t;
    }

    // --- суффикс 'f' / 'F' → кадры ---
    if (last == 'f' || last == 'F') {
        double v = 0.0;
        if (!parse_double(s.substr(0, s.size() - 1), v)) return std::nullopt;
        if (v <= 0.0) return std::nullopt;
        // Дробное число кадров не имеет физического смысла:
        // движок вставляет/удаляет целые кадры.
        if (v != std::floor(v)) return std::nullopt;
        TargetSpec t;
        t.kind  = TargetSpec::Kind::Frames;
        t.value = v;
        return t;
    }

    // --- чистое число → секунды ---
    double v = 0.0;
    if (!parse_double(s, v)) return std::nullopt;
    if (v <= 0.0) return std::nullopt;
    TargetSpec t;
    t.kind  = TargetSpec::Kind::Seconds;
    t.value = v;
    return t;
}

// ---------------------------------------------------------------------------
// Разбор аргументов командной строки.
// ---------------------------------------------------------------------------
std::expected<Args, Error>
parse_args(int argc, char** argv, std::string_view prg) {
    Args a;

    // Единые точки выхода для -h и -V: используются и в ранней
    // проверке argv, и в switch ниже (для слитных форм -hXXX/-VXXX).
    auto emit_help = [&]() -> std::expected<Args, Error> {
        print_usage(stdout, prg);
        return std::unexpected(Error::Help);
    };
    auto emit_version = [&]() -> std::expected<Args, Error> {
        print("{} {}\n", prg, kVersion);
        return std::unexpected(Error::Version);
    };

    // Единая точка выхода для «неверных аргументов»: печатает
    // опциональное сообщение, usage и возвращает InvalidArgs.
    // Устраняет дублирование print_usage(stderr, prg) + return
    // по всему телу parse_args().
    auto fail_usage = [&](std::string_view msg = {})
            -> std::expected<Args, Error> {
        if (!msg.empty()) println(stderr, "{}", msg);
        print_usage(stderr, prg);
        return std::unexpected(Error::InvalidArgs);
    };

    if (argc < 2) return fail_usage();

    bool have_infile = false;

    for (int i = 1; i < argc; ++i) {
        const std::string_view v = argv[i];

        // "--" завершает список опций: все последующие аргументы —
        // позиционные (пути), даже если начинаются с '-'.
        if (v == "--") {
            for (++i; i < argc; ++i) {
                if (have_infile)
                    return fail_usage("Multiple input files are not supported.");
                a.infile    = argv[i];
                have_infile = true;
            }
            break;
        }

        if (v == "-h" || v == "--help" || v == "/?"
#ifdef _WIN32
            || v == "/h"
#endif
            ) {
            return emit_help();
        }

        // -V / --version: печатаем версию в stdout и выходим.
        // Обработка перенесена из main.cpp сюда, чтобы сторонние
        // вызывающие parse_args() получили единообразное поведение.
        if (v == "-V" || v == "--version") {
            return emit_version();
        }

        // Определяем, опция это или путь. На Unix '/' — начало абсолютного
        // пути, а не префикс опции; учитываем это.
        if (v == "-")
            return fail_usage("Reading from stdin ('-') is not supported.");

        bool is_opt = (!v.empty() && v[0] == '-');
#ifdef _WIN32
        // На Windows исторически опции могли начинаться с '/' (cmd.exe),
        // но в MSYS2 путь тоже начинается с '/'. Поэтому распознаём '/'
        // как префикс опции только для известных форм: '/h', '/?',
        // '/t', '/o' и их слитных вариантов ('/t20.5'). Пути вида
        // '/tmp/...' не должны интерпретироваться как опции.
        if (!is_opt && v.size() >= 2 && v[0] == '/' && v[1] != '/') {
            const char c = v[1];
            if (c == '?') {
                is_opt = true;
            } else if (c == 'h') {
                is_opt = (v.size() == 2);
            } else if (c == 't') {
                is_opt = (v.size() == 2)
                      || (v.size() > 2
                          && ((v[2] >= '0' && v[2] <= '9') || v[2] == '.'));
            } else if (c == 'o') {
                is_opt = (v.size() == 2);
            }
        }
#endif

        if (is_opt) {
            // Опция может быть слитной: -t20.5, -oout.mp2.
	    const char opt = v[1];  // v.size() >= 2 — is_opt check above guarantees it
            const std::string_view optarg =
                v.size() > 2 ? v.substr(2) : std::string_view{};

            // Возвращает значение опции: либо уже слитое,
            // либо следующий argv-элемент.
            auto need_value = [&](std::string_view& out) -> bool {
                if (!optarg.empty()) { out = optarg; return true; }
                if (i + 1 >= argc) return false;
                out = argv[++i];
                return true;
            };

            switch (opt) {
            // Точные формы -h/--help/-V/--version перехвачены выше,
            // до определения is_opt. Эти case-ветки достижимы только
            // для слитных форм (-hXXX, -VXXX): у флагов нет аргументов,
            // поэтому хвост optarg игнорируется.
            case 'h':
                return emit_help();
            case 'V':
                return emit_version();
            case 't': {
                std::string_view tval;
                if (!need_value(tval))
                    return fail_usage("Option -t requires a value.");
                auto spec = parse_target(tval);
                if (!spec)
                    return fail_usage(std::format("Invalid -t value: {}", tval));
                a.target = *spec;
                break;
            }
            case 'o': {
                std::string_view oval;
                if (!need_value(oval))
                    return fail_usage("Option -o requires a value.");
                a.outfile.assign(oval);
                break;
            }
            default:
                return fail_usage();
            }
        } else {
            if (have_infile)
                return fail_usage("Multiple input files are not supported.");
            a.infile    = argv[i];
            have_infile = true;
        }
    }

    if (!have_infile) return fail_usage();
    return a;
}

// ---------------------------------------------------------------------------
// Печать времени HH:MM:SS.mmm
// ---------------------------------------------------------------------------
void print_time(double t) {
    if (!(t > 0.0)) t = 0.0;
    // Округляем до миллисекунд ДО разбиения на компоненты,
    // чтобы избежать ситуации 59.9995 → "60.000" в младшем
    // поле при неизменных минутах/часах.
    const long long total_ms =
        static_cast<long long>(t * 1000.0 + 0.5);
    const long long h  = total_ms / 3600000LL;
    const long long m  = (total_ms / 60000LL) % 60LL;
    const long long s  = (total_ms / 1000LL) % 60LL;
    const long long ms = total_ms % 1000LL;
    print("{:02}:{:02}:{:02}.{:03}", h, m, s, ms);
}

// ---------------------------------------------------------------------------
// Детектор формата: DTS → AC-3 → MP2. DTS пробуется первым, потому что
// 0x7FFE8001 — самое длинное синхрослово и случайное совпадение маловероятно.
// ---------------------------------------------------------------------------
std::optional<Detected> detect_codec(std::span<const std::uint8_t> head,
                                     std::size_t total_size) noexcept {
    const Codec* order[3] = { &dts_codec(), &ac3_codec(), &mp2_codec() };

    // Pass 1: strict — следующий sync обязателен.
    for (const Codec* c : order) {
        auto r = find_frame(head, *c, total_size, false);
        if (r) {
            Detected d;
            d.codec  = c;
            d.offset = r->first;
            d.fi     = r->second;
            return d;
        }
    }

    // Pass 2: relax — одиночный кадр в начале + хвостовой мусор.
    for (const Codec* c : order) {
        auto r = find_frame(head, *c, total_size, true);
        if (r) {
            Detected d;
            d.codec  = c;
            d.offset = r->first;
            d.fi     = r->second;
            return d;
        }
    }

    return std::nullopt;
}

// ---------------------------------------------------------------------------
// Полный цикл: детект, анализ, план, конвейер.
// ---------------------------------------------------------------------------
std::expected<void, Error>
scale_stream(const Args& args) {
    // -------- входной файл --------
    FilePtr fin(std::fopen(args.infile.c_str(), "rb"), &std::fclose);
    if (!fin) {
        println(stderr, "File {} could not be opened", args.infile);
        return std::unexpected(Error::FileOpen);
    }

    // -------- размер файла --------
    std::error_code ec;
    const auto fsz_opt = std::filesystem::file_size(args.infile, ec);
    if (ec) return std::unexpected(Error::FileSeek);
    const std::size_t fsize = static_cast<std::size_t>(fsz_opt);

    if (fsize < 10) {
        println(stderr, "File {} is too short.", args.infile);
        return std::unexpected(Error::TooShort);
    }
    std::rewind(fin.get());

    // -------- 1. Детект формата --------
    const std::size_t head_len = std::min<std::size_t>(fsize, kScanBuf);
    std::vector<std::uint8_t> head(head_len);
    if (std::fread(head.data(), 1, head_len, fin.get()) != head_len)
        return std::unexpected(Error::FileRead);

    auto det = detect_codec(std::span<const std::uint8_t>{head}, fsize);
    if (!det) {
        print(stderr,
              "File {} is not a recognized audio stream. "
              "Supported codecs: {}, {}, {}.\n",
              args.infile,
              mp2_codec().name(), ac3_codec().name(), dts_codec().name());
        return std::unexpected(Error::BadFormat);
    }

    const Codec&    codec  = *det->codec;
    const std::size_t njunk = det->offset;
    const FrameInfo fi0     = det->fi;

    codec.print_info(std::span<const std::uint8_t>{head}.subspan(njunk), fi0);

    // -------- 2. Хвостовой мусор --------
    const std::size_t tail_len = std::min<std::size_t>(fsize, kBufSize);
    std::vector<std::uint8_t> tail(tail_len);
    if (SCALE_FSEEK(fin.get(),
                    static_cast<std::int64_t>(fsize - tail_len),
                    SEEK_SET) != 0)
        return std::unexpected(Error::FileSeek);
    if (std::fread(tail.data(), 1, tail_len, fin.get()) != tail_len)
        return std::unexpected(Error::FileRead);

    const std::size_t ntrail =
        tail_junk_size(std::span<const std::uint8_t>{tail}, codec);

    if (fsize < njunk + ntrail) return std::unexpected(Error::TooShort);
    const std::size_t flen = fsize - njunk - ntrail;

    // -------- 3. Точный проход по потоку --------
    ScanResult scan = scan_frames(fin.get(), njunk, flen, codec);
    if (!scan.ok) {
        print(stderr,
              "Failed to scan input stream: stream ends prematurely.\n");
        return std::unexpected(Error::FileRead);
    }
    if (scan.nframes == 0) {
        println(stderr, "No valid frames found in input stream.");
        return std::unexpected(Error::BadFormat);
    }

    const std::size_t nframes = scan.nframes;
    const double      fsec    = scan.total_sec;
    // nframes > 0 and each framesec > 0 are guaranteed by
    // parse_header() for all supported codecs.
    const double      avg_sec = fsec / static_cast<double>(nframes);

    print("Found {} leading and {} trailing junk bytes.\n", njunk, ntrail);
    print("Original length of audio is {} seconds, or ", fsec);
    print_time(fsec);
    print(", in {} frames\n", nframes);

    // -------- 4. Целевая длительность и план --------
    double target_sec = 0.0;
    if (args.target.active()) {
        target_sec = (args.target.kind == TargetSpec::Kind::Frames)
            ? args.target.value * avg_sec
            : args.target.value;
    }

    // delta — целое число кадров, которое нужно вставить (delta > 0)
    // или удалить (delta < 0). Округление до целого гарантирует,
    // что итоговое число действий совпадёт с расчётным.
    long delta = 0;
    if (target_sec > 0.0 && nframes > 0 && avg_sec > 0.0) {
        delta = std::lround((target_sec - fsec) / avg_sec);
        print("Approximately {} frames need to be ", std::labs(delta));
        if (delta > 0)      print("inserted.\n");
        else if (delta < 0) print("dropped.\n");
        else                print("nothing to do.\n");
    }

    if (args.outfile.empty()) return {};   // только отчёт

    if (!args.target.active()) {
        if (njunk > 0 || ntrail > 0) {
            print(stderr,
                  "Note: -o without -t; leading/trailing junk "
                  "will be stripped.\n");
        } else if (codec.has_extension()) {
            print(stderr,
                  "Note: -o without -t; output is DTS Core "
                  "(HD extensions, if present, will be dropped).\n");
        } else {
            print(stderr,
                  "Note: -o without -t; output will be an exact copy.\n");
        }
    }

    // -------- 5. Построение плана и открытие выхода --------
    // Отказываемся перезаписывать входной файл: fopen("wb")
    // усечёт его до того, как мы закончим чтение.
    {
        std::error_code eqec;
                bool refuse = false;
        if (std::filesystem::exists(args.outfile, eqec) && !eqec) {
            const bool eq = std::filesystem::equivalent(
                args.infile, args.outfile, eqec);
            refuse = (!eqec) ? eq : true;  // error -> refuse
        }
        if (refuse) {
            print(stderr,
                  "Refusing to overwrite input file {} with output.
",
                  args.outfile);
            return std::unexpected(Error::FileOpen);
        }
    }

    // Сначала строим план. Если построение упадёт (std::bad_alloc),
    // выходной файл ещё не будет усечён.
    std::vector<Action> plan(nframes, Action::Copy);
    if (delta != 0 && nframes > 0) {
        const long absd = std::labs(delta);
        if (absd >= static_cast<long>(nframes)) {
            // Target unattainable in one pass.
            if (delta < 0) {
                // Dropping all frames yields an empty (0-frame) file.
                // Keep at least one frame; distribute drops evenly.
                const long keep = static_cast<long>(nframes) - 1;
                print(stderr,
                      "Warning: target requires {} drops but only {} frames "
                      "are available; dropping {} (keeping at least one "
                      "frame); result will be approximate.\n",
                      absd, nframes, keep);
                distribute_actions(plan, nframes, keep, Action::Drop);
            } else {
                print(stderr,
                      "Warning: target requires {} insertions but only {} "
                      "frames are available; inserting one duplicate per "
                      "frame; result will be approximate.\n",
                      absd, nframes);
                // distribute_actions с count == nframes вырождается
                // в «пометить все позиции» — ровно то, что нужно.
                distribute_actions(plan, nframes,
                                   static_cast<long>(nframes), Action::Dup);
            }
        } else {
            distribute_actions(plan, nframes, absd,
                               (delta > 0) ? Action::Dup : Action::Drop);
        }
    }

    FilePtr fout(std::fopen(args.outfile.c_str(), "wb"), &std::fclose);
    if (!fout) {
        println(stderr, "File {} could not be opened", args.outfile);
        return std::unexpected(Error::FileOpen);
    }

    // -------- 6. Конвейер reader/writer --------
    BoundedQueue<FrameBlock> q(kQueueCap);
    std::atomic<bool>        reader_ok{true};

    std::jthread reader([&](std::stop_token st) {
        const ReadStatus rs = for_each_frame(fin.get(), njunk, flen, codec,
            [&](std::span<const std::uint8_t> bytes, const FrameInfo& fi) {
                FrameBlock fb;
                fb.bytes.assign(bytes.begin(), bytes.end());
                fb.framesec = fi.framesec;
                return q.push(std::move(fb));
            }, st);

        if (rs == ReadStatus::Error)
            reader_ok.store(false, std::memory_order_relaxed);
        q.close();
    });

    double      written_sec = 0.0;
    std::size_t nfread      = 0;
    long        nf          = 0;
    bool        write_ok    = true;

    while (auto item = q.pop()) {
        if (nfread >= nframes) break;
        const Action a = plan[nfread];

        // Проверяем результат fwrite: переполнение диска и
        // ошибки ввода-вывода должны прерывать обработку, а не
        // молча давать усечённый выходной файл.
        auto put = [&](const std::vector<std::uint8_t>& bytes) -> bool {
            return std::fwrite(bytes.data(), 1, bytes.size(),
                               fout.get()) == bytes.size();
        };

        switch (a) {
            case Action::Copy:
                if (!put(item->bytes)) write_ok = false;
                written_sec += item->framesec;
                break;
            case Action::Dup:
                if (!put(item->bytes)) write_ok = false;
                if (!put(item->bytes)) write_ok = false;
                written_sec += 2.0 * item->framesec;
                ++nf;
                break;
            case Action::Drop:
                --nf;
                break;
        }
        ++nfread;
    }

    reader.request_stop();
    q.close();   // разблокировать reader, если он застрял в push()

    if (!reader_ok.load(std::memory_order_relaxed))
        return std::unexpected(Error::FileRead);

    // Сброс буфера и проверка отложенных ошибок записи:
    // переполнение диска часто проявляется только на flush.
    if (write_ok && std::fflush(fout.get()) != 0) write_ok = false;
    if (!write_ok)
        return std::unexpected(Error::FileWrite);

    if (nfread != nframes) {
        // Reader не смог выдать все кадры. Это не должно
        // происходить при корректном flen, но лучше сообщить
        // явно, чем молча выдать усечённый файл.
        print(stderr, "Warning: processed only {} of {} frames.\n",
              nfread, nframes);
    }

    print("{}: ", codec.long_name());
    if (delta > 0)      print("Inserted {} frames, ", nf);
    else if (delta < 0) print("Removed {} frames, ", -nf);
    else                print("No frames changed, ");
    print("actual length now is ");
    print_time(written_sec);
    print(" in {} frames.\n", static_cast<long>(nfread) + nf);

    return {};
}

} // namespace scale

/*
   dtscodec.cpp — реализация scale::Codec для DTS Coherent Acoustics.

   Distributed under GNU GPL v2 or later.

   Класс DtsCodec инкапсулирует всё, что относится к DTS Core, а также
   автоматическое извлечение DTS Core из DTS-HD MA/HRA:

     - синхрослово DTS Core (16-bit BE = 0x7FFE8001, 16-bit LE = 0xFE7F0180);
     - разбор заголовка кадра (NBLKS, FSIZE, AMODE, SFREQ, BITRATE);
     - вычисление размера кадра и его длительности;
     - переопределённый frame_span(): сканирует следующий DTS Core sync,
       что автоматически отбрасывает XLL/X96-расширение DTS-HD и даёт на
       выходе чистый DTS Core;
     - печать человекочитаемой сводки о потоке.

   Таблицы соответствуют ETSI TS 102 114 (DTS Coherent Acoustics).

   Формат заголовка DTS Core (после 4-байтового sync, MSB-first):
     Frame Type                (1)
     Deficit Sample Count      (5)
     CRC Present               (1)
     Number of PCM Sample Blocks (7) [+1, 1..128]        → NBLKS
     Primary Frame Byte Size   (14) [+1, 1..16384]       → FSIZE
     Audio Channel Arrangement (6)                        → AMODE
     Core Audio Sampling Frequency (4)                    → SFREQ
     Transmission Bit Rate     (5)                        → BITRATE

   Кадры DTS Core самодостаточны, но в DTS-HD MA/HRA сразу за ядром идёт
   extension (XLL для MA, X96 для HRA) со своим sync-словом. Мы находим
   следующий DTS Core sync после ядра — всё, что между ними, является
   extension'ом и отбрасывается. Так ядро извлекается «бесплатно».
*/

#include "scale/common.hpp"

#include <array>
#include <cstddef>
#include <cstdint>
#include <expected>
#include <optional>
#include <span>
#include <string_view>

namespace {

// Единый набор using-деклараций, чтобы не квалифицировать имена
// в каждой сигнатуре.
using scale::print;
using scale::Error;
using scale::FrameInfo;
using scale::detail::read_be32;
using scale::detail::kNone;

// ---------------------------------------------------------------------------
// Таблицы DTS Core (ETSI TS 102 114)
// ---------------------------------------------------------------------------

// SFREQ (4 бита) -> частота дискретизации, Гц.
//   0x0 invalid, 0x1 8k, 0x2 16k, 0x3 32k, 0x4 64k, 0x5 128k,
//   0x6 11025, 0x7 22050, 0x8 44100, 0x9 88200, 0xA 176400,
//   0xB 12000, 0xC 24000, 0xD 48000, 0xE 96000, 0xF 192000.
constexpr std::array<double, 16> kFreqs = {
    0.0,      8000.0,   16000.0,  32000.0,
    64000.0,  128000.0, 11025.0,  22050.0,
    44100.0,  88200.0,  176400.0, 12000.0,
    24000.0,  48000.0,  96000.0,  192000.0
};

// BITRATE (5 бит) -> битрейт, кбит/с.
//   0x00..0x04 reserved, 0x05 = 128, ..., 0x18 = 1536,
//   0x19..0x1F reserved / user-defined.
constexpr std::array<std::size_t, 32> kBitrates = {
    0,    0,    0,    0,    0,      // 0x00..0x04 reserved
    128,  192,  224,  256,  320,    // 0x05..0x09
    384,  448,  512,  576,  640,    // 0x0A..0x0E
    768,  960,  1024, 1152, 1280,   // 0x0F..0x13
    1344, 1408, 1411, 1472, 1536,   // 0x14..0x18
    0,    0,    0,    0,    0,      // 0x19..0x1D reserved
    0,    0                          // 0x1E..0x1F user-defined
};

// AMODE[amode & 0x0F] -> число каналов (без LFE).
// Источник: FFmpeg libavcodec/dcadata.h, ff_dca_channels[16].
//   LFE (amode & 0x10) учитывается отдельно.
constexpr std::array<std::size_t, 16> kAmodeChannels = {
    1, 2, 2, 2, 2, 3, 3, 4, 4, 5, 6, 6, 6, 7, 8, 8
};

// Минимальный размер кадра DTS Core (FSIZE = 96, нижняя граница стандарта).
constexpr std::size_t kMinFrameBytes = 96;

// Максимальный размер кадра DTS Core: FSIZE = 14 бит, +1 → 16384.
constexpr std::size_t kMaxFrameBytes = 16384;

// ---------------------------------------------------------------------------
// Вспомогательные функции для побитового чтения MSB-first.
// ---------------------------------------------------------------------------


// Извлечение N бит, начиная с битовой позиции bitpos (MSB-first),
// из массива байт. Используется для полей заголовка, не выровненных
// по байтовой границе. Вызывающий обязан гарантировать, что
// (bitpos + n + 7) / 8 <= b.size().
[[nodiscard]] constexpr std::uint32_t
read_bits(std::span<const std::uint8_t> b, std::size_t bitpos, unsigned n) noexcept {
    std::uint32_t v = 0;
    for (unsigned i = 0; i < n; ++i) {
        const std::size_t bp = bitpos + i;
        const std::size_t byte = bp >> 3;
        const unsigned   bit  = 7u - static_cast<unsigned>(bp & 7u);
        v = (v << 1) | ((b[byte] >> bit) & 1u);
    }
    return v;
}

// ---------------------------------------------------------------------------
// Реализация scale::Codec для DTS Core.
// ---------------------------------------------------------------------------

class DtsCodec final : public scale::Codec {
public:
    // -----------------------------------------------------------------------
    // Метаданные
    // -----------------------------------------------------------------------
    [[nodiscard]] std::string_view name()      const noexcept override { return "dts"; }
    [[nodiscard]] std::string_view long_name() const noexcept override { return "DTS"; }

    // -----------------------------------------------------------------------
    // Синхрослово DTS Core.
    //   16-bit BE: 0x7FFE8001
    //   16-bit LE: 0xFE7F0180 (байты sync-swap'нуты по 16-битным словам)
    // 14-bit BE (0x1FFFE800, DTS-in-WAV) намеренно не поддерживается для
    // парсинга — заголовок в нём хранится в 14-битных словах, что требует
    // отдельного битового ридера. Такие файлы лучше предварительно
    // конвертировать в 16-bit BE.
    // -----------------------------------------------------------------------
    [[nodiscard]] bool
    has_sync(std::span<const std::uint8_t> b) const noexcept override {
        if (b.size() < 4) return false;
        const std::uint32_t v = read_be32(b);
        return v == 0x7FFE8001u   // 16-bit BE
            || v == 0xFE7F0180u;  // 16-bit LE
    }

    // -----------------------------------------------------------------------
    // Разбор заголовка кадра DTS Core.
    // Требуется минимум 10 байт: 4 sync + 6 байт заголовка (43 бита).
    // -----------------------------------------------------------------------
    [[nodiscard]] std::expected<FrameInfo, Error>
    parse_header(std::span<const std::uint8_t> b) const noexcept override {
        const auto hdr = parse_core_header(b);
        if (!hdr) return std::unexpected(Error::BadFormat);

        const std::size_t nblks = hdr->nblks;
        const std::size_t fsize = hdr->fsize;

        if (kFreqs[hdr->sfreq] == 0.0)
            return std::unexpected(Error::BadFormat);
        if (kBitrates[hdr->brate] == 0)
            return std::unexpected(Error::BadFormat);
        if (fsize < kMinFrameBytes || fsize > kMaxFrameBytes)
            return std::unexpected(Error::BadFormat);

        const std::size_t samples_per_frame = nblks * 32;

        FrameInfo fi{};
        fi.framebytes   = fsize;
        fi.framesec     = static_cast<double>(samples_per_frame)
                        / kFreqs[hdr->sfreq];
        fi.freq_khz     = kFreqs[hdr->sfreq] / 1000.0;
        fi.bitrate_kbps = kBitrates[hdr->brate];
        return fi;
    }

    // -----------------------------------------------------------------------
    // frame_span() — ключевое отличие DTS от MP2/AC-3.
    //
    // Для MP2/AC-3 базовый Codec::frame_span() возвращает fi.framebytes,
    // и это корректно. Для DTS Core в DTS-HD MA/HRA сразу за ядром идёт
    // extension (XLL/X96) со своим sync'ом; следующий DTS Core sync
    // находится дальше. Найдя его, мы возвращаем полное расстояние от
    // текущего sync до следующего, а читающий код копирует только
    // fi.framebytes байт (само ядро) — extension отбрасывается.
    //
    //   * Fast path: если следующий sync стоит сразу за ядром (чистый
    //     DTS Core), возвращаем core.
    //   * Slow path: сканируем буфер начиная с core+1 до ближайшего sync.
    //   * Если sync не найден — возвращаем SIZE_MAX, движок должен
    //     дозаполнить буфер.
    // -----------------------------------------------------------------------
    [[nodiscard]] std::size_t
    frame_span(std::span<const std::uint8_t> buf,
               const scale::FrameInfo& fi) const noexcept override {
        const std::size_t core = fi.framebytes;

        if (core + 4 > buf.size()) return kNone;
        if (has_sync(buf.subspan(core))) return core;

        for (std::size_t j = core + 1; j + 10 <= buf.size(); ++j) {
            auto view = buf.subspan(j);
            if (has_sync(view) && parse_header(view)) return j;
        }
        return kNone;
    }

    // -----------------------------------------------------------------------
    // DTS-HD MA/HRA содержит XLL/X96 extension после ядра,
    // поэтому проверка «следующий sync сразу за ядром» неприменима.
    // -----------------------------------------------------------------------
    [[nodiscard]] bool has_extension() const noexcept override {
        return true;
    }

    // -----------------------------------------------------------------------
    // Человекочитаемая сводка о потоке.
    // -----------------------------------------------------------------------
    void print_info(std::span<const std::uint8_t> b,
                    const scale::FrameInfo& fi) const override {
        const auto hdr = parse_core_header(b);
        if (!hdr) return;

        const std::size_t nch = kAmodeChannels[hdr->amode & 0x0F];
        const bool        lfe = (hdr->amode & 0x10) != 0;

        print("Audio type is DTS, {}, {} channel(s){}, {} samples per frame, "
              "sampling freq is {} kHz, and bitrate is {} kbps.\n",
              hdr->is_16le ? "16-bit LE" : "16-bit BE",
              nch, lfe ? ".1" : "",
              hdr->nblks * 32,
              fi.freq_khz, fi.bitrate_kbps);
    }

private:
    // -----------------------------------------------------------------------
    // Единый разбор 4-байтного sync + 6-байтного заголовка DTS Core.
    // Возвращает nullopt, если sync не распознан или буфер слишком мал.
    //
    // Для 16-bit LE байты 16-битных слов идут в обратном порядке —
    // разворачиваем их во временный массив внутри структуры, чтобы
    // возвращаемый span не висел на локальном буфере.
    // -----------------------------------------------------------------------
    struct CoreHeader {
        bool        is_16le = false;
        std::size_t nblks   = 0;   // 1..128
        std::size_t fsize   = 0;   // 1..16384
        std::size_t amode   = 0;   // 6 бит
        std::size_t sfreq   = 0;   // 4 бита
        std::size_t brate   = 0;   // 5 бит
    };

    [[nodiscard]] static std::optional<CoreHeader>
    parse_core_header(std::span<const std::uint8_t> b) noexcept {
        if (b.size() < 10) return std::nullopt;

        const std::uint32_t sync = read_be32(b);
        const bool is_16be = (sync == 0x7FFE8001u);
        const bool is_16le = (sync == 0xFE7F0180u);
        if (!is_16be && !is_16le) return std::nullopt;

        std::array<std::uint8_t, 6> hdr_swap{};
        std::span<const std::uint8_t> hdr;
        if (is_16be) {
            hdr = b.subspan(4, 6);
        } else {
            for (std::size_t i = 0; i < 6; i += 2) {
                hdr_swap[i]     = b[4 + i + 1];
                hdr_swap[i + 1] = b[4 + i];
            }
            hdr = hdr_swap;
        }

        CoreHeader out;
        out.is_16le = is_16le;
        out.nblks   = read_bits(hdr,  7,  7) + 1;
        out.fsize   = read_bits(hdr, 14, 14) + 1;
        out.amode   = read_bits(hdr, 28,  6);
        out.sfreq   = read_bits(hdr, 34,  4);
        out.brate   = read_bits(hdr, 38,  5);
        return out;
    }
};

} // namespace

// Фабрика
// ---------------------------------------------------------------------------

const scale::Codec& scale::dts_codec() {
    static const DtsCodec instance;
    return instance;
}

/*
   scale — единая утилита масштабирования MP2 / AC-3 / DTS Core.

   Distributed under GNU GPL v2 or later.

   Единственная точка входа проекта. Формат входного файла определяется
   автоматически (см. scale::detect_codec()):

     * MP2 (MPEG-1 Layer II)                  — масштабируется как есть;
     * AC-3 (Dolby Digital)                   — масштабируется как есть;
     * DTS Core                               — масштабируется как есть;
     * DTS-HD MA / HRA                        — извлекается DTS Core,
                                                extension отбрасывается.

   Использование:
     scale [-t target] [-o outfile] infile
     scale --help

   Формат target (-t):
     20.5        20.5 секунд (по умолчанию секунды);
     20.5s       20.5 секунд (явно);
     500f        500 кадров;
     01:30       1 минута 30 секунд (mm:ss);
     00:20:30    20 минут 30 секунд (hh:mm:ss).

   Без -t выводится только сводка о входном потоке.
   Без -o масштабирование не производится (аналог "dry run").
*/

#include "scale/common.hpp"

#include <string_view>

namespace {

// ---------------------------------------------------------------------------
// Единая точка выхода для ошибок. Для Error::Help и Error::Version
// возвращаем 0 (пользователь сам запросил справку/версию — не ошибка),
// иначе печатаем краткую сводку в stderr и возвращаем 1. Детальные
// сообщения уже выведены parse_args() / scale_stream().
[[nodiscard]] int exit_with(std::string_view prg,
                            scale::Error      e,
                            std::string_view why) noexcept {
    if (e == scale::Error::Help || e == scale::Error::Version)
        return 0;
    scale::print(stderr, "{}: {}\n", prg, why);
    return 1;
}

} // namespace

int main(int argc, char** argv) {
    // Имя программы для строки использования. Берём из argv[0], отбрасывая
    // путь — это делает вывод аккуратным при вызове "./scale" или "C:\\...\\scale.exe".
    std::string_view prg = (argc > 0 && argv[0] != nullptr) ? argv[0] : "scale";
    if (auto pos = prg.find_last_of("/\\"); pos != std::string_view::npos)
        prg.remove_prefix(pos + 1);
    if (prg.empty()) prg = "scale";

    // -----------------------------------------------------------------------
    // 1. Разбор аргументов.
    //    parse_args() возвращает Error::Help при явном запросе
    //    справки (-h/--help), Error::Version — при -V/--version (версия
    //    уже напечатана), Error::InvalidArgs — при некорректных опциях.
    // -----------------------------------------------------------------------
    auto args = scale::parse_args(argc, argv, prg);
    if (!args) {
        const scale::Error e = args.error();
        // Если пользователь явно запросил справку (-h / --help), parse_args
        // возвращает Help, а не InvalidArgs — выходим с кодом 0.
        return exit_with(prg, e, scale::to_string(e));
    }

    // -----------------------------------------------------------------------
    // 2. Полный цикл: детект формата, анализ, план dup/drop,
    //    при необходимости — запись результата в outfile.
    //    Формат входного файла выбирается автоматически; DTS-HD MA/HRA
    //    превращается в DTS Core на лету (см. dtscodec.cpp:frame_span()).
    // -----------------------------------------------------------------------
    auto result = scale::scale_stream(*args);
    if (!result)
        return exit_with(prg, result.error(), scale::to_string(result.error()));

    return 0;
}

/*
   mp2codec.cpp — реализация scale::Codec для MPEG-1 Layer II.

   Copyright (c) 2004, by Zhuo Meng (zhuo@thunder.cwru.edu).
   Distributed under GNU GPL v2 or later.

   Класс Mp2Codec инкапсулирует всё, что относится к формату MP2:
     - синхрослово 0xFFF и его проверку;
     - разбор заголовка кадра (layer, битрейт, частота, padding, CRC);
     - вычисление размера кадра и его длительности;
     - печать человекочитаемой сводки о потоке.

   Важно:
     * никаких статических глобальных буферов — всё работает через
       std::span, границы проверяются на стороне вызывающего (common.cpp);
     * размер кадра может меняться на 1 байт из-за бита padding, поэтому
       движок обязан вызывать parse_header() на каждом кадре — базовый
       Codec::frame_span() возвращает fi.framebytes, что здесь корректно,
       так как parse_header уже учёл padding в framebytes.
*/

#include "scale/common.hpp"

#include <array>
#include <cstddef>
#include <cstdint>
#include <expected>
#include <span>
#include <string_view>

namespace {

// Единый набор using-деклараций, чтобы не квалифицировать имена в каждой
// сигнатуре.
using scale::print;
using scale::Error;
using scale::FrameInfo;

// ---------------------------------------------------------------------------
// Таблицы MPEG-1 Layer II (ISO/IEC 11172-3)
// ---------------------------------------------------------------------------

// Индекс битрейта (4 бита) -> кбит/с.
//   0000 = free, 1111 = reserved/bad. Оба случая отклоняются.
constexpr std::array<std::size_t, 16> kBitrates = {
    0,     // 0000 — free
    32, 48, 56, 64, 80, 96, 112, 128, 160, 192, 224, 256, 320, 384,
    0      // 1111 — reserved / bad
};

// Индекс частоты (2 бита) -> кГц. 3 => reserved.
// Порядок для MPEG-1: 44.1, 48, 32. Оригинальные утилиты поддерживали
// только MPEG-1 — сохраняем это поведение.
constexpr std::array<double, 4> kFreqs = { 44.1, 48.0, 32.0, 0.0 };

// Число сэмплов на кадр для Layer II (MPEG-1).
constexpr std::size_t kSamplesPerFrame = 1152;

// Минимально допустимый размер кадра — отсекаем мусор.
constexpr std::size_t kMinFrameBytes = 4;

// ---------------------------------------------------------------------------
// Реализация scale::Codec для MP2
// ---------------------------------------------------------------------------

class Mp2Codec final : public scale::Codec {
public:
    // -----------------------------------------------------------------------
    // Метаданные
    // -----------------------------------------------------------------------
    [[nodiscard]] std::string_view name()      const noexcept override { return "mp2"; }
    [[nodiscard]] std::string_view long_name() const noexcept override { return "MP2"; }

    // -----------------------------------------------------------------------
    // Синхрослово: 11 бит "1111 1111 111".
    // На практике достаточно b[0] == 0xFF и старших 4 бит второго байта.
    // -----------------------------------------------------------------------
    [[nodiscard]] bool
    has_sync(std::span<const std::uint8_t> b) const noexcept override {
        return b.size() >= 2
            && b[0] == 0xFF
            && (b[1] & 0xF0) == 0xF0;
    }

    // -----------------------------------------------------------------------
    // Разбор заголовка кадра.
    // Требуется минимум 4 байта (заголовок MPEG-1 Layer II без CRC).
    //
    // Раскладка байтов (MSB-first):
    //   b[0]: 11111111                — sync (11 бит)
    //   b[1]: 111VVLLP                — VV=версия, LL=layer, P=protection
    //   b[2]: BBBBFFFP                — BBBB=битрейт, FFF=частота, P=padding
    //   b[3]: MMCC....                — MM=channel mode, CC=mode extension
    // -----------------------------------------------------------------------
    [[nodiscard]] std::expected<FrameInfo, Error>
    parse_header(std::span<const std::uint8_t> b) const noexcept override {
        // Синхрослово обязательно: иначе мусор мог бы пройти как
        // заголовок Layer II (см. потерю sync в for_each_frame).
        if (b.size() < 4 || !has_sync(b))
            return std::unexpected(Error::BadFormat);

        const std::uint8_t b1 = b[1];
        const std::uint8_t b2 = b[2];

        // Layer (биты 2..1): 01=Layer III, 10=Layer II, 11=Layer I.
        // Нас интересует только Layer II.
        const int layer = 4 - ((b1 & 0x06) >> 1);
        if (layer != 2)
            return std::unexpected(Error::BadFormat);

        // Таблицы ниже — для MPEG-1. MPEG-2/2.5 Layer II использует
        // другой набор битрейтов; такой поток здесь не поддержан.
        if (((b1 >> 3) & 0x03) != 0x03)
            return std::unexpected(Error::BadFormat);

        // Индексы битрейта / частоты / padding.
        const std::size_t brInd   =
            static_cast<std::size_t>((b2 & 0xF0u) >> 4);
        const std::size_t freqInd =
            static_cast<std::size_t>((b2 & 0x0Cu) >> 2);
        const bool        pad     = (b2 & 0x02) != 0;

        // Валидация индексов — иначе OOB по массивам и деление на ноль.
        if (kBitrates[brInd] == 0)
            return std::unexpected(Error::BadFormat);
        if (freqInd >= 3 || kFreqs[freqInd] == 0.0)
            return std::unexpected(Error::BadFormat);

        // Формула размера кадра MPEG-1 Layer II
        // (битрейт в кбит/с, частота в кГц):
        //     frame_size = bitrate * 144 / sample_rate + padding
        // где 144 = 1152 сэмпла / 8 бит * 1000 (единицы кбит/с, кГц).
        //
        // Вычисление в double, чтобы избежать -Wconversion при приведении
        // size_t → double на этапе деления на kFreqs[freqInd].
        const double frame_size_f =
            static_cast<double>(kBitrates[brInd]) * 144.0 / kFreqs[freqInd];
        const std::size_t framebytes =
            static_cast<std::size_t>(frame_size_f) + (pad ? 1u : 0u);

        if (framebytes < kMinFrameBytes)
            return std::unexpected(Error::BadFormat);

        FrameInfo fi{};
        fi.framebytes   = framebytes;
        fi.framesec     = static_cast<double>(kSamplesPerFrame)
                        / (kFreqs[freqInd] * 1000.0);
        fi.freq_khz     = kFreqs[freqInd];
        fi.bitrate_kbps = kBitrates[brInd];
        return fi;
    }

    // -----------------------------------------------------------------------
    // frame_span() для MP2 не переопределяем.
    //
    // После корректного парсинга framebytes уже включает padding, поэтому
    // следующий sync находится ровно на fi.framebytes от текущего. Базовая
    // реализация Codec::frame_span() возвращает именно fi.framebytes —
    // этого достаточно.
    // -----------------------------------------------------------------------

    // -----------------------------------------------------------------------
    // Человекочитаемая сводка о потоке.
    // -----------------------------------------------------------------------
    void print_info(std::span<const std::uint8_t> b,
                    const scale::FrameInfo& fi) const override {
        if (b.size() < 4) return;

        const std::uint8_t b1 = b[1];
        const std::uint8_t b3 = b[3];

        // MPEG audio version: 00=2.5, 01=reserved, 10=MPEG-2, 11=MPEG-1.
        // Оригинал работал только с MPEG-1, но формат сводки сохраняем.
        // MPEG audio version (bits 4..3 of b1):
        //   00 = MPEG-2.5, 01 = reserved, 10 = MPEG-2, 11 = MPEG-1
        static constexpr std::array<const char*, 4> kMpegVerNames = {
            "2.5", "reserved", "2", "1"
        };
        const std::size_t verBits = (b1 >> 3) & 0x03;
        const char* mpgv = kMpegVerNames[verBits];
        const int protect  = 1 - (b1 & 0x01);          // 1 => есть CRC

        // Channel mode (биты 7..6 третьего байта):
        //   00 stereo, 01 joint stereo, 10 dual channel, 11 mono.
        const int chanmode = (b3 & 0xC0) >> 6;

        print("Audio type is MPEG-{}, Layer 2, ", mpgv);
        switch (chanmode) {
            case 0: print("stereo, ");       break;
            case 1: print("joint stereo, "); break;
            case 2: print("dual channel, "); break;
            case 3: print("mono, ");         break;
            default: break;
        }
        print("{}, each frame is {} bytes, sampling freq is {} kHz, "
              "and bitrate is {} kbps.\n",
              protect ? "with CRC" : "without CRC",
              fi.framebytes, fi.freq_khz, fi.bitrate_kbps);
    }
};

} // namespace

// ---------------------------------------------------------------------------
// Фабрика
// ---------------------------------------------------------------------------

const scale::Codec& scale::mp2_codec() {
    static const Mp2Codec instance;
    return instance;
}

// ---------------------------------------------------------------------------
// Общие хелперы для синтетических кадров.
// Используются как в test_detect.cpp, так и в test_roundtrip.cpp,
// чтобы не дублировать битовую раскладку заголовков.
// ---------------------------------------------------------------------------

// MPEG-1, Layer II, 192 кбит/с, 48 кГц, без CRC, stereo.
// frame_size = 192 * 144 / 48 = 576 байт.
inline std::vector<std::uint8_t> make_mp2_frame() {
    std::vector<std::uint8_t> f(576, 0x00);
    f[0] = 0xFF;
    f[1] = 0xFD;
    f[2] = 0xA4;
    f[3] = 0x00;
    return f;
}

// N последовательных MP2-кадров.
inline std::vector<std::uint8_t>
make_mp2_stream(std::size_t nframes) {
    // Строится из make_mp2_frame() — единственный источник
    // истины для байтовой раскладки заголовка MP2.
    const auto frame = make_mp2_frame();
    std::vector<std::uint8_t> data;
    data.reserve(nframes * frame.size());
    for (std::size_t i = 0; i < nframes; ++i)
        data.insert(data.end(), frame.begin(), frame.end());
    return data;
}

// AC-3: 48 kHz, frmsizecod=20 → 384 слова → 768 байт (192 кбит/с).
inline std::vector<std::uint8_t> make_ac3_frame() {
    std::vector<std::uint8_t> f(768, 0x00);
    f[0] = 0x0B; f[1] = 0x77;   // sync
    f[4] = 0x14;                // fscod=0 (48 kHz), frmsizecod=20
    f[5] = 0x40;                // bsid=8
    return f;
}

// N последовательных AC-3-кадров.
inline std::vector<std::uint8_t>
make_ac3_stream(std::size_t nframes) {
    const auto frame = make_ac3_frame();
    std::vector<std::uint8_t> data;
    data.reserve(nframes * frame.size());
    for (std::size_t i = 0; i < nframes; ++i)
        data.insert(data.end(), frame.begin(), frame.end());
    return data;
}

// DTS Core: 48 kHz, 1536 кбит/с, NBLKS=16, FSIZE=96, AMODE=2.
inline std::vector<std::uint8_t> make_dts_frame() {
    std::vector<std::uint8_t> f(96, 0x00);
    f[0] = 0x7F; f[1] = 0xFE; f[2] = 0x80; f[3] = 0x01;
    f[4] = 0x02; f[5] = 0x3C; f[6] = 0x05;
    f[7] = 0xF0; f[8] = 0xB7; f[9] = 0x00;
    return f;
}

// N последовательных DTS Core кадров.
inline std::vector<std::uint8_t>
make_dts_stream(std::size_t nframes) {
    const auto frame = make_dts_frame();
    std::vector<std::uint8_t> data;
    data.reserve(nframes * frame.size());
    for (std::size_t i = 0; i < nframes; ++i)
        data.insert(data.end(), frame.begin(), frame.end());

// DTS-HD: ядро DTS Core + сигнатура XLL. frame_span() должен перепрыгнуть
// extension и вернуть расстояние до следующего ядра.
SCALE_TEST(dts_frame_span_skips_hd_extension) {
    auto core1 = make_dts_frame();
    auto core2 = make_dts_frame();

    std::vector<std::uint8_t> xll;
    xll.push_back(0x41); xll.push_back(0xA2);
    xll.push_back(0x95); xll.push_back(0x47);
    xll.insert(xll.end(), 60, 0x00);

    std::vector<std::uint8_t> buf;
    buf.insert(buf.end(), core1.begin(), core1.end());
    buf.insert(buf.end(), xll.begin(),   xll.end());
    buf.insert(buf.end(), core2.begin(), core2.end());

    const Codec& c = dts_codec();
    auto fi = c.parse_header(buf);
    SCALE_CHECK(fi.has_value());
    SCALE_CHECK_EQ(fi->framebytes, std::size_t{96});

    const std::size_t span = c.frame_span(buf, *fi);
    SCALE_CHECK_EQ(span, std::size_t{96} + xll.size());
}

#!/usr/bin/env bash
# =============================================================================
#  build-msys2-ucrt64.sh — сборка scale в MSYS2 UCRT64
#  с полной статической линковкой и глубокой оптимизацией под amd64/Windows 10+.
#
#  Запускать из MSYS2 UCRT64 shell:
#      ./build-msys2-ucrt64.sh
#
#  Требования:
#    - MSYS2 с окружением UCRT64 (переменная MSYSTEM=UCRT64);
#    - mingw-w64-ucrt-x86_64-gcc, mingw-w64-ucrt-x86_64-cmake,
#      mingw-w64-ucrt-x86_64-ninja (устанавливаются автоматически).
# =============================================================================

set -euo pipefail

# --- Проверка окружения MSYS2 UCRT64 ---------------------------------------
if [[ "${MSYSTEM:-}" != "UCRT64" ]]; then
    echo "Ошибка: скрипт нужно запускать из MSYS2 UCRT64 shell." >&2
    echo "Откройте 'MSYS2 UCRT64' из меню Пуск и повторите." >&2
    exit 1
fi

# --- Установка зависимостей через pacman ------------------------------------
# Список пакетов: компилятор, CMake, Ninja. Git не нужен для сборки.
PACMAN_PKGS=(
    mingw-w64-ucrt-x86_64-gcc
    mingw-w64-ucrt-x86_64-cmake
    mingw-w64-ucrt-x86_64-ninja
)

# Проверяем, установлены ли уже нужные пакеты (по наличию cmake и ninja).
if ! command -v cmake >/dev/null 2>&1 \
   || ! command -v ninja >/dev/null 2>&1 \
   || ! command -v g++   >/dev/null 2>&1; then
    echo ">> Устанавливаю зависимости через pacman..."
    pacman -S --needed --noconfirm "${PACMAN_PKGS[@]}"
fi

# --- Параметры сборки --------------------------------------------------------
BUILD_DIR="${BUILD_DIR:-build-ucrt64-static}"
BUILD_TYPE="${BUILD_TYPE:-Release}"

# Флаги архитектуры и Release-профиля задаются в CMakeLists.txt
# (SCALE_NATIVE=OFF → -march=x86-64-v3, SCALE_NATIVE=ON → -march=native).
# Здесь оставляем переменные пустыми, чтобы не дублировать -march/-mtune.
ARCH_FLAGS=""

# Дополнительные флаги компиляции (точечные переопределения).
EXTRA_CXX_FLAGS=""

# Флаги линкера: полная статика + удаление неиспользуемых секций.
# -static заставляет линковать статически libgcc, libstdc++ и winpthread.
# Если в будущем понадобится оставить какую-то системную DLL динамической,
# замените -static на -static-libgcc -static-libstdc++ и добавьте нужные
# исключения.
LINKER_FLAGS="-static -static-libgcc -static-libstdc++ \
-Wl,--gc-sections -Wl,-O1"

# --- Конфигурация и сборка ---------------------------------------------------
echo ">> Конфигурирую CMake (${BUILD_TYPE}) в ${BUILD_DIR}..."
cmake -S . -B "${BUILD_DIR}" \
    -G Ninja \
    -DCMAKE_BUILD_TYPE="${BUILD_TYPE}" \
    -DCMAKE_CXX_FLAGS="${ARCH_FLAGS} ${EXTRA_CXX_FLAGS}" \
    -DCMAKE_EXE_LINKER_FLAGS="${LINKER_FLAGS}" \
    -DSCALE_NATIVE=OFF \
    -DSCALE_WERROR=OFF \
    -DSCALE_BUILD_TESTS=OFF

echo ">> Собираю..."
cmake --build "${BUILD_DIR}" --parallel "$(nproc)"

echo ">> Готово. Бинарник:"
ls -lh "${BUILD_DIR}/scale.exe" 2>/dev/null || ls -lh "${BUILD_DIR}/scale" 2>/dev/null || echo "   (не найден — проверьте ${BUILD_DIR})"

.PARAMETER Arch

.PARAMETER Pgo
    Включить PGO-цикл. Требует -Sample.

.PARAMETER Sample
    Путь к тренировочному аудиофайлу (.mp2 / .ac3 / .dts / .dtshd).

.PARAMETER Target
    Целевая длительность в секундах для PGO-прогона. По умолчанию 10.

.PARAMETER Native
    Собрать с оптимизацией под текущий CPU (аналог -march=native).
    Для MSVC маппится в /arch:AVX2 (у MSVC нет прямого аналога).

.PARAMETER Werror
    Считать предупреждения ошибками (/WX).

.PARAMETER Tests
    Собрать и запустить CTest-тесты.

.PARAMETER Clean
    Удалить каталоги сборки и выйти.

.PARAMETER BuildDir
    Каталог сборки. По умолчанию "build".

.EXAMPLE
    .\build.ps1
    .\build.ps1 -Configuration Debug -Ninja
    .\build.ps1 -Pgo -Sample .\sample.dts -Target 20.5
    .\build.ps1 -Tests

#>

    [ValidateSet("Debug","Release","RelWithDebInfo","MinSizeRel")]
    [string] $Configuration = "Release",

    [switch] $Ninja,
    [switch] $Pgo,
    [switch] $Native,

    [string] $BuildDir = "build"
)

# -----------------------------------------------------------------------------
# Строгий режим и остановка на первой ошибке
# -----------------------------------------------------------------------------
Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

# -----------------------------------------------------------------------------
# Единая точка правды: имя единственного бинарника проекта
# -----------------------------------------------------------------------------
$MainExe = "scale"

# -----------------------------------------------------------------------------
# Цветной вывод
# -----------------------------------------------------------------------------
function Write-Log     ($m) { Write-Host ">> $m"  -ForegroundColor Cyan }
function Write-Ok      ($m) { Write-Host "OK $m"  -ForegroundColor Green }
function Write-WarnMsg ($m) { Write-Warning  $m }
function Write-Err     ($m) { Write-Host "!! $m"  -ForegroundColor Red }

function Die ($m) {
    Write-Err $m
    exit 1
}

# -----------------------------------------------------------------------------
# Проверка окружения
# -----------------------------------------------------------------------------
function Test-Command ($name) {
    return [bool](Get-Command $name -ErrorAction SilentlyContinue)
}

if (-not (Test-Command "cmake")) {
    Die "cmake не найден в PATH. Установите CMake 3.25+ или добавьте в PATH."
}

$cmakeVersion = (& cmake --version | Select-Object -First 1) -replace '^cmake version\s+',''
Write-Log "cmake $cmakeVersion"

# Проверка ОС: для _WIN32_WINNT=0x0A00 нужна Windows 10+.
$osVer = [System.Environment]::OSVersion.Version
if ($osVer.Major -lt 10) {
    Write-WarnMsg "Обнаружена ОС ниже Windows 10 ($($osVer)). Сборка продолжится, но _WIN32_WINNT=0x0A00 может быть некорректен."
}

# -----------------------------------------------------------------------------
# Чистка
# -----------------------------------------------------------------------------
if ($Clean) {
    Write-Log "Удаляю каталоги сборки: $BuildDir, build-pgo"
    foreach ($d in @($BuildDir, "build-pgo")) {
        if (Test-Path $d) {
            Remove-Item -Recurse -Force -LiteralPath $d
        }
    }
    Write-Ok "Готово"
    exit 0
}

# -----------------------------------------------------------------------------
# Число параллельных задач
# -----------------------------------------------------------------------------
if (-not $env:JOBS) {
    $Jobs = [Environment]::ProcessorCount
    if ($Jobs -le 0) { $Jobs = 4 }
} else {
    $Jobs = [int] $env:JOBS
}
Write-Log "jobs=$Jobs"

# -----------------------------------------------------------------------------
# Выбор генератора
# -----------------------------------------------------------------------------
if ($Ninja) {
    if (-not ((Test-Command "cl") -or (Test-Command "clang-cl") -or (Test-Command "clang++"))) {
        Die "Ни cl.exe, ни clang-cl.exe, ни clang++.exe не найдены в PATH. Запустите скрипт из 'x64 Native Tools Command Prompt for VS 2022'."
    }
    if (-not (Test-Command "ninja")) {
        Die "ninja не найден в PATH. Установите Ninja или используйте VS-генератор."
    }
    $Generator     = "Ninja"
    $IsMultiConfig = $false
} else {
    # Ищем VS 2022; при отсутствии — падаем с понятным сообщением.
    $vsWhere = Join-Path ${env:ProgramFiles(x86)} "Microsoft Visual Studio\Installer\vswhere.exe"
    $vsPath  = $null
    if (Test-Path $vsWhere) {
        $vsPath = & $vsWhere -latest -products * `
                             -requires Microsoft.VisualStudio.Component.VC.Tools.x86.x64 `
                             -property installationPath 2>$null
    }
    if (-not $vsPath) {
        Die "Visual Studio 2022 с C++ toolchain не найден. Установите VS 2022 или используйте -Ninja."
    }
    $Generator     = "Visual Studio 17 2022"
    $IsMultiConfig = $true
}
Write-Log "generator: $Generator"

# -----------------------------------------------------------------------------
# Общие cmake-флаги
# -----------------------------------------------------------------------------
$commonArgs = @(
    "-S", ".",
    "-B", $BuildDir,
    "-G", $Generator,
    "-DCMAKE_BUILD_TYPE=$Configuration",
    "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON",
    "-DSCALE_NATIVE=$(if ($Native) {'ON'} else {'OFF'})",
    "-DSCALE_WERROR=$(if ($Werror) {'ON'} else {'OFF'})",
    "-DSCALE_BUILD_TESTS=$(if ($Tests)  {'ON'} else {'OFF'})"
)

if (-not $Ninja) {
    $commonArgs += @("-A", $Arch)
}

# -----------------------------------------------------------------------------
# Поиск собранного exe с учётом multi-config
# -----------------------------------------------------------------------------
function Get-ExePath ([string]$base, [string]$cfg, [string]$name) {
    $candidates = @(
        (Join-Path $base "$name.exe"),
        (Join-Path $base "$cfg\$name.exe"),
        (Join-Path $base "bin\$name.exe"),
        (Join-Path $base "bin\$cfg\$name.exe")
    )
    foreach ($c in $candidates) {
        if (Test-Path $c -PathType Leaf) { return $c }
    }
    return $null
}

# -----------------------------------------------------------------------------
# Печать артефактов
# -----------------------------------------------------------------------------
function Write-Artifacts ([string]$base, [string]$cfg, [string]$label = "Артефакты") {
    Write-Host ""
    Write-Log "${label}:"
    $p = Get-ExePath -base $base -cfg $cfg -name $MainExe
    if ($p) {
        $size = "{0:N0} KiB" -f ((Get-Item $p).Length / 1KB)
        Write-Host ("  {0,-12} {1,10}  {2}" -f $MainExe, $size, $p)
    } else {
        Write-WarnMsg "Не найден бинарник $MainExe"
    }
}

# =============================================================================
# Обычная (не-PGO) сборка
# =============================================================================
if (-not $Pgo) {
    Write-Log "Конфигурирую $Configuration ($Arch) в $BuildDir"
    & cmake @commonArgs
    if ($LASTEXITCODE -ne 0) { Die "cmake configure не удался" }

    Write-Log "Собираю (parallel=$Jobs)"
    & cmake --build $BuildDir --config $Configuration --parallel $Jobs
    if ($LASTEXITCODE -ne 0) { Die "cmake build не удался" }

    Write-Ok "Сборка завершена"

    Write-Artifacts -base $BuildDir -cfg $Configuration

    if ($Tests) {
        Write-Log "Запускаю CTest"
        $ctestArgs = @("--test-dir", $BuildDir, "--output-on-failure",
                       "--parallel", "$Jobs")
        if ($IsMultiConfig) { $ctestArgs += @("-C", $Configuration) }
        & ctest @ctestArgs
        if ($LASTEXITCODE -ne 0) { Die "Тесты упали" }
        Write-Ok "Тесты прошли"
    }

    exit 0
}

# =============================================================================
# PGO-цикл: generate → run → use
#
# Инструментированная и финальная сборки выполняются в ОДНОМ каталоге
# ($pgoDir): GCC и Clang встраивают в .gcda/.profraw пути к объектным
# файлам, и при смене каталога профиль не находится.
# =============================================================================
Write-Log "PGO-режим включён"

if ([string]::IsNullOrWhiteSpace($Sample)) {
    Die "Для PGO укажите -Sample <файл.mp2|файл.ac3|файл.dts|файл.dtshd>"
}
if (-not (Test-Path $Sample -PathType Leaf)) {
    Die "Sample-файл не найден: $Sample"
}
$Sample = (Resolve-Path $Sample).Path

Write-Log "Тренировочный бинарник: $MainExe (формат определится автоматически)"

# Инструментированная и финальная сборки выполняются в одном каталоге.
$pgoDir  = "build-pgo"
$profDir = Join-Path $pgoDir "prof"
if (-not (Test-Path $profDir)) { New-Item -ItemType Directory -Path $profDir | Out-Null }
$profDirFull = (Resolve-Path $profDir).Path

# -----------------------------------------------------------------------------
# Определяем инструмент компиляции: MSVC cl.exe или clang-cl.exe.
# -----------------------------------------------------------------------------
$useClangCl = $false
if ($Ninja) {
    if ($env:CXX) {
        # Пользователь явно задал компилятор — уважаем выбор.
        $useClangCl = ($env:CXX -match 'clang')
    } elseif ((Test-Command "clang-cl") -and -not (Test-Command "cl")) {
        # cl.exe имеет приоритет как «родной» MSVC-компилятор.
        # clang-cl используется, только если cl.exe недоступен.
        $useClangCl = $true
    }
}

# -----------------------------------------------------------------------------
# Формируем флаги PGO.
# -----------------------------------------------------------------------------
if ($useClangCl) {
    Write-Log "PGO через clang-cl (-fprofile-instr-generate / -fprofile-instr-use)"

    # -fprofile-instr-generate не принимает каталог: конкретный .profraw
    # задаётся переменной LLVM_PROFILE_FILE во время прогона.
    $genCompileFlags  = "-fprofile-instr-generate"
    $genLinkerFlags   = "-fprofile-instr-generate"

    # Значения use-флагов формируются после merge .profraw → .profdata.
    $useCompileFlags  = ""
    $useLinkerFlags   = ""
} else {
    Write-Log "PGO через MSVC (/GL → прогон → /LTCG:PGO)"

    $env:VCPROFILE_PATH = $profDirFull

    $genCompileFlags  = "/GL"
    $genLinkerFlags   = "/LTCG:PGI"

    $useCompileFlags  = "/GL"
    $useLinkerFlags   = "/LTCG:PGO"
}

# --- Шаг 1: сборка с инструментированием -------------------------------------
Write-Log "[PGO 1/3] Сборка с инструментированием в $pgoDir"
$genArgs = @(
    "-S", ".", "-B", $pgoDir, "-G", $Generator,
    "-DCMAKE_BUILD_TYPE=$Configuration",
    "-DCMAKE_CXX_FLAGS=$genCompileFlags",
    "-DCMAKE_EXE_LINKER_FLAGS=$genLinkerFlags",
    "-DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF",
    "-DSCALE_PGO=ON",
    "-DSCALE_NATIVE=$(if ($Native) {'ON'} else {'OFF'})",
    "-DSCALE_WERROR=OFF",
    "-DSCALE_BUILD_TESTS=OFF"
)
if (-not $Ninja) { $genArgs += @("-A", $Arch) }

& cmake @genArgs
if ($LASTEXITCODE -ne 0) { Die "cmake configure (PGO gen) не удался" }

& cmake --build $pgoDir --config $Configuration --parallel $Jobs
if ($LASTEXITCODE -ne 0) { Die "cmake build (PGO gen) не удался" }

# --- Шаг 2: прогон на sample ------------------------------------------------
Write-Log "[PGO 2/3] Прогон на $Sample (target=$Target s)"
$sampleOut = Join-Path ([System.IO.Path]::GetTempPath()) `
                       ("pgo_" + [guid]::NewGuid().ToString("N") + ".out")

function Invoke-Train {
    $path = Get-ExePath -base $pgoDir -cfg $Configuration -name $MainExe
    if (-not $path) { Die "Не найден тренировочный бинарник $MainExe" }

    foreach ($i in 1..3) {
        if ($useClangCl) {
            $env:LLVM_PROFILE_FILE = Join-Path $profDirFull "run${i}_%p.profraw"
        }
        & $path -t $Target -o $sampleOut $Sample *> $null
        if ($LASTEXITCODE -ne 0) {
            Write-WarnMsg "Прогон $MainExe #$i завершился с кодом $LASTEXITCODE"
        }
    }
    if ($useClangCl) {
        $env:LLVM_PROFILE_FILE = Join-Path $profDirFull "report_%p.profraw"
    }
    & $path $Sample *> $null

    Write-Ok "Профиль для $MainExe собран"
}

try {
    Invoke-Train
} finally {
    if (Test-Path $sampleOut) { Remove-Item -Force $sampleOut }
    if ($useClangCl) { Remove-Item Env:LLVM_PROFILE_FILE -ErrorAction SilentlyContinue }
}

# --- Шаг 2.5 (только clang-cl): merge .profraw → .profdata ------------------
if ($useClangCl) {
    if (-not (Test-Command "llvm-profdata")) {
        Die "llvm-profdata не найден. Установите llvm-tools."
    }
    $profData = Join-Path $profDirFull "merged.profdata"
    $rawFiles = @(Get-ChildItem -Path $profDirFull -Filter *.profraw -ErrorAction SilentlyContinue)
    if ($rawFiles.Count -eq 0) {
        Die "Не найдено ни одного .profraw в $profDirFull"
    }
    Write-Log "Объединяю $($rawFiles.Count) .profraw → $profData"
    & llvm-profdata merge -o $profData @($rawFiles.FullName)
    if ($LASTEXITCODE -ne 0) { Die "llvm-profdata merge не удался" }
    $useCompileFlags = "-fprofile-instr-use=`"$profData`" -fprofile-correction"
    $useLinkerFlags  = "-fprofile-instr-use=`"$profData`""
}

# --- Шаг 3: финальная сборка с профилем -------------------------------------
Write-Log "[PGO 3/3] Финальная сборка с профилем в $pgoDir"
$useArgs = @(
    "-S", ".", "-B", $pgoDir, "-G", $Generator,
    "-DCMAKE_BUILD_TYPE=$Configuration",
    "-DCMAKE_CXX_FLAGS=$useCompileFlags",
    "-DCMAKE_EXE_LINKER_FLAGS=$useLinkerFlags",
    "-DSCALE_PGO=ON",
    "-DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF",
    "-DSCALE_NATIVE=$(if ($Native) {'ON'} else {'OFF'})",
    "-DSCALE_WERROR=$(if ($Werror) {'ON'} else {'OFF'})",
    "-DSCALE_BUILD_TESTS=$(if ($Tests)  {'ON'} else {'OFF'})"
)
if (-not $Ninja) { $useArgs += @("-A", $Arch) }

& cmake @useArgs
if ($LASTEXITCODE -ne 0) { Die "cmake configure (PGO use) не удался" }

& cmake --build $pgoDir --config $Configuration --parallel $Jobs
if ($LASTEXITCODE -ne 0) { Die "cmake build (PGO use) не удался" }

Write-Ok "PGO-сборка завершена"

Write-Artifacts -base $pgoDir -cfg $Configuration -label "Артефакты (PGO)"

# -----------------------------------------------------------------------------
# Опциональное сравнение с baseline (BENCH=1)
# -----------------------------------------------------------------------------
if ($env:BENCH) {
    Write-Log "Сравнение времени работы baseline vs PGO:"
    $baseDir  = "build-baseline"
    $basePath = Get-ExePath -base $baseDir -cfg $Configuration -name $MainExe
    if (-not $basePath) {
        Write-Log "Baseline не найден — собираю $baseDir (без PGO)"
        $baseArgs = @(
            "-S", ".", "-B", $baseDir, "-G", $Generator,
            "-DCMAKE_BUILD_TYPE=$Configuration",
            "-DSCALE_PGO=OFF",
            "-DSCALE_NATIVE=$(if ($Native) {'ON'} else {'OFF'})",
            "-DSCALE_WERROR=OFF",
            "-DSCALE_BUILD_TESTS=OFF"
        )
        if (-not $Ninja) { $baseArgs += @("-A", $Arch) }
        & cmake @baseArgs
        if ($LASTEXITCODE -ne 0) { Die "cmake configure (baseline) не удался" }
        & cmake --build $baseDir --config $Configuration --parallel $Jobs
        if ($LASTEXITCODE -ne 0) { Die "cmake build (baseline) не удался" }
        $basePath = Get-ExePath -base $baseDir -cfg $Configuration -name $MainExe
    }
    $pgoPath = Get-ExePath -base $pgoDir -cfg $Configuration -name $MainExe
    if ($pgoPath -and $basePath) {
        $t1 = (Measure-Command { & $basePath $Sample *> $null }).TotalMilliseconds
        $t2 = (Measure-Command { & $pgoPath  $Sample *> $null }).TotalMilliseconds
        $speedup = if ($t2 -gt 0) { "{0:N2}x" -f ($t1 / $t2) } else { "n/a" }
        Write-Host ("  {0,-12} baseline={1,8:N0} ms  pgo={2,8:N0} ms  speedup={3}" -f `
                     $MainExe, $t1, $t2, $speedup)
    } else {
        Write-WarnMsg "Не удалось получить пути к baseline/pgo бинарникам"
    }
}

exit 0
--- КОНЕЦ ФАЙЛА: ./build.ps1 ---

--- НАЧАЛО ФАЙЛА: ./build.sh [TYPE: Text] [ENCODING: UTF-8] [LINE-ENDING: LF (Unix)] ---
#!/usr/bin/env bash
# =============================================================================
#  build.sh — кроссплатформенная сборка утилиты scale
#             (MP2 / AC-3 / DTS Core, включая извлечение ядра из DTS-HD MA/HRA).
#
#  Возможности:
#    * выбор генератора (Ninja / Makefiles) и build-type;
#    * автоматическое определение числа ядер;
#    * опциональный PGO-цикл (generate → run → use);
#    * поддержка SCALE_NATIVE / SCALE_WERROR / SCALE_BUILD_TESTS;
#    * чистка build-каталогов и вывод артефактов.
#
#  Использование:
#    ./build.sh                    # Release-сборка в ./build
#    ./build.sh Debug              # Debug-сборка
#    ./build.sh Release --pgo --sample sample.ac3 --target 20.5
#    ./build.sh --clean            # удалить build-каталоги
#    ./build.sh --native           # -march=native
#    ./build.sh --werror           # -Werror
#    ./build.sh --tests            # собрать и запустить unit-тесты
#
#  Переменные окружения:
#    CXX=clang++       — выбор компилятора
#    BUILD_DIR=build   — каталог сборки
#    JOBS=8            — число параллельных задач
# =============================================================================

set -euo pipefail

# -----------------------------------------------------------------------------
# Цветной вывод
# -----------------------------------------------------------------------------
if [[ -t 1 ]] && command -v tput >/dev/null 2>&1 \
   && [[ "$(tput colors 2>/dev/null || echo 0)" -ge 8 ]]; then
    C_RESET="$(tput sgr0)"
    C_BOLD="$(tput bold)"
    C_RED="$(tput setaf 1)"
    C_GREEN="$(tput setaf 2)"
    C_YELLOW="$(tput setaf 3)"
    C_CYAN="$(tput setaf 6)"
else
    C_RESET="" C_BOLD="" C_RED="" C_GREEN="" C_YELLOW="" C_CYAN=""
fi

log()   { printf '%s>>%s %s\n'  "${C_CYAN}${C_BOLD}"   "${C_RESET}" "$*"; }
ok()    { printf '%s✓%s  %s\n'  "${C_GREEN}${C_BOLD}"  "${C_RESET}" "$*"; }
warn()  { printf '%s!%s  %s\n'  "${C_YELLOW}${C_BOLD}" "${C_RESET}" "$*" >&2; }
err()   { printf '%s✗%s  %s\n'  "${C_RED}${C_BOLD}"    "${C_RESET}" "$*" >&2; }
die()   { err "$*"; exit 1; }

# -----------------------------------------------------------------------------
# Единая точка правды для артефактов
# -----------------------------------------------------------------------------
MAIN_EXE="scale"

# -----------------------------------------------------------------------------
# Разбор аргументов
# -----------------------------------------------------------------------------
BUILD_TYPE="Release"
BUILD_DIR="${BUILD_DIR:-build}"
PGO=0
CLEAN=0
NATIVE=0
WERROR=0
TESTS=0
GENERATOR=""
SAMPLE_INPUT="${SAMPLE_INPUT:-}"
SAMPLE_TARGET="${SAMPLE_TARGET:-10.0}"

while [[ $# -gt 0 ]]; do
    case "$1" in
        Debug|Release|RelWithDebInfo|MinSizeRel)
            BUILD_TYPE="$1"; shift ;;
        --pgo)        PGO=1; shift ;;
        --clean)      CLEAN=1; shift ;;
        --native)     NATIVE=1; shift ;;
        --werror)     WERROR=1; shift ;;
        --tests)      TESTS=1; shift ;;
        --ninja)      GENERATOR="Ninja"; shift ;;
        --make)       GENERATOR="Unix Makefiles"; shift ;;
        --build-dir)  BUILD_DIR="$2"; shift 2 ;;
        --sample)     SAMPLE_INPUT="$2"; shift 2 ;;
        --target)     SAMPLE_TARGET="$2"; shift 2 ;;
        -h|--help)
            sed -n '2,30p' "$0" | sed 's/^# \{0,1\}//'
            exit 0 ;;
        *) die "Неизвестный аргумент: $1 (см. --help)" ;;
    esac
done

# -----------------------------------------------------------------------------
# Чистка
# -----------------------------------------------------------------------------
if [[ "$CLEAN" -eq 1 ]]; then
    log "Удаляю каталог сборки: ${BUILD_DIR}"
    rm -rf -- "$BUILD_DIR"
    ok "Готово"
    exit 0
fi

# -----------------------------------------------------------------------------
# Определение числа параллельных задач
# -----------------------------------------------------------------------------
if [[ -z "${JOBS:-}" ]]; then
    if command -v nproc >/dev/null 2>&1; then
        JOBS="$(nproc)"
    elif command -v sysctl >/dev/null 2>&1; then
        JOBS="$(sysctl -n hw.ncpu 2>/dev/null || echo 4)"
    elif command -v getconf >/dev/null 2>&1; then
        JOBS="$(getconf _NPROCESSORS_ONLN 2>/dev/null || echo 4)"
    else
        JOBS=4
    fi
fi

# -----------------------------------------------------------------------------
# Проверка инструментов
# -----------------------------------------------------------------------------
command -v cmake >/dev/null 2>&1 || die "cmake не найден в PATH"
CMAKE_VERSION="$(cmake --version | head -n1 | awk '{print $3}')"
log "cmake ${CMAKE_VERSION}"

CXX_BIN="${CXX:-c++}"
command -v "$CXX_BIN" >/dev/null 2>&1 || die "C++ компилятор '$CXX_BIN' не найден"
CXX_VERSION="$("$CXX_BIN" --version | head -n1)"
log "C++ compiler: ${CXX_VERSION}"

# Автовыбор генератора, если не задан явно.
if [[ -z "$GENERATOR" ]]; then
    if command -v ninja >/dev/null 2>&1; then
        GENERATOR="Ninja"
    else
        GENERATOR="Unix Makefiles"
    fi
fi
log "generator: ${GENERATOR}"

# -----------------------------------------------------------------------------
# Общие cmake-флаги
# -----------------------------------------------------------------------------
COMMON_FLAGS=(
    -DCMAKE_BUILD_TYPE="${BUILD_TYPE}"
    -DCMAKE_CXX_COMPILER="${CXX_BIN}"
    -DCMAKE_EXPORT_COMPILE_COMMANDS=ON
    -DSCALE_NATIVE="$([[ "$NATIVE" -eq 1 ]] && echo ON || echo OFF)"
    -DSCALE_WERROR="$([[ "$WERROR" -eq 1 ]] && echo ON || echo OFF)"
    -DSCALE_BUILD_TESTS="$([[ "$TESTS" -eq 1 ]] && echo ON || echo OFF)"
)

# -----------------------------------------------------------------------------
# Печать артефактов: ищет exe во всех возможных местах
# -----------------------------------------------------------------------------
print_artifacts() {
    local base="$1"
    local cfg="$2"
    local label="${3:-Артефакты}"
    echo
    log "${label}:"
    local cand
    for cand in \
        "${base}/${MAIN_EXE}" \
        "${base}/${cfg}/${MAIN_EXE}" \
        "${base}/bin/${MAIN_EXE}" \
        "${base}/bin/${cfg}/${MAIN_EXE}"; do
        if [[ -x "$cand" ]]; then
            local size
            size="$(du -h "$cand" | cut -f1)"
            printf '  %s%-20s%s %s  %s\n' \
                "$C_BOLD" "$MAIN_EXE" "$C_RESET" "$size" "$cand"
            return
        fi
    done
    warn "Не найден бинарник ${MAIN_EXE}"
}

# =============================================================================
# Обычная (не-PGO) сборка
# =============================================================================
if [[ "$PGO" -eq 0 ]]; then
    log "Конфигурирую ${BUILD_TYPE} в ${BUILD_DIR} (jobs=${JOBS})"
    cmake -S . -B "$BUILD_DIR" -G "$GENERATOR" "${COMMON_FLAGS[@]}"

    log "Собираю"
    cmake --build "$BUILD_DIR" --parallel "$JOBS"

    ok "Сборка завершена"

    print_artifacts "$BUILD_DIR" "$BUILD_TYPE"

    # -------------------------------------------------------------------------
    # Быстрая самопроверка (если запрошены тесты)
    # -------------------------------------------------------------------------
    if [[ "$TESTS" -eq 1 ]]; then
        log "Запускаю CTest"
        ctest --test-dir "$BUILD_DIR" --output-on-failure --parallel "$JOBS"
        ok "Тесты прошли"
    fi

    exit 0
fi

# =============================================================================
# PGO-цикл: generate → run → use
#
# Инструментированная и финальная сборки выполняются в ОДНОМ каталоге
# ($BUILD_DIR): GCC и Clang встраивают в .gcda/.profraw пути к объектным
# файлам, и при смене каталога профиль не находится.
# =============================================================================
log "PGO-режим включён"

[[ -n "$SAMPLE_INPUT" ]] || die "Для PGO укажите --sample <файл.mp2|файл.ac3|файл.dts>"
[[ -f "$SAMPLE_INPUT" ]]  || die "Sample-файл не найден: $SAMPLE_INPUT"

PROF_DIR="${BUILD_DIR}/prof"
mkdir -p "$PROF_DIR"

log "Тренировочный бинарник: ${MAIN_EXE} (формат определяется автоматически)"

IS_CLANG=0
[[ "$CXX_BIN" == *clang* ]] && IS_CLANG=1

# --- Шаг 1: сборка с инструментированием -------------------------------------
log "[PGO 1/3] Сборка с инструментированием в ${BUILD_DIR}"
if [[ "$IS_CLANG" -eq 1 ]]; then
    # Clang: -fprofile-instr-generate без пути; конкретное имя .profraw
    # задаётся переменной LLVM_PROFILE_FILE во время прогона.
    PGO_GEN_FLAGS=(-fprofile-instr-generate)
    PGO_GEN_LDFLAGS=(-fprofile-instr-generate)
else
    # GCC: -fprofile-generate=<dir> задаёт каталог для .gcda.
    PGO_GEN_FLAGS=(-fprofile-generate="${PWD}/${PROF_DIR}")
    PGO_GEN_LDFLAGS=(-fprofile-generate="${PWD}/${PROF_DIR}")
fi

cmake -S . -B "$BUILD_DIR" -G "$GENERATOR" \
    "${COMMON_FLAGS[@]}" \
    -DSCALE_PGO=ON \
    -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF \
    -DCMAKE_CXX_FLAGS="${PGO_GEN_FLAGS[*]}" \
    -DCMAKE_EXE_LINKER_FLAGS="${PGO_GEN_LDFLAGS[*]}"

cmake --build "$BUILD_DIR" --parallel "$JOBS"

# --- Шаг 2: прогон на sample-файле -------------------------------------------
log "[PGO 2/3] Прогон на ${SAMPLE_INPUT} (target=${SAMPLE_TARGET}s)"
SAMPLE_OUT="$(mktemp -t pgo_XXXXXX.out)"
trap 'rm -f "$SAMPLE_OUT"' EXIT

run_train() {
    local path="$BUILD_DIR/$MAIN_EXE"
    [[ -x "$path" ]] || path="$BUILD_DIR/$BUILD_TYPE/$MAIN_EXE"
    [[ -x "$path" ]] || die "Не найден тренировочный бинарник: $MAIN_EXE"

    local i
    if [[ "$IS_CLANG" -eq 1 ]]; then
        for i in 1 2 3; do
            LLVM_PROFILE_FILE="${PWD}/${PROF_DIR}/run${i}_%p.profraw" \
                "$path" -t "$SAMPLE_TARGET" -o "$SAMPLE_OUT" "$SAMPLE_INPUT" >/dev/null
        done
        LLVM_PROFILE_FILE="${PWD}/${PROF_DIR}/report_%p.profraw" \
            "$path" "$SAMPLE_INPUT" >/dev/null
    else
        for i in 1 2 3; do
            "$path" -t "$SAMPLE_TARGET" -o "$SAMPLE_OUT" "$SAMPLE_INPUT" >/dev/null
        done
        "$path" "$SAMPLE_INPUT" >/dev/null
    fi
    ok "Профиль для $MAIN_EXE собран"
}

run_train

# --- Шаг 3: финальная сборка с использованием профиля ------------------------
log "[PGO 3/3] Финальная сборка с профилем в ${BUILD_DIR}"
if [[ "$IS_CLANG" -eq 1 ]]; then
    command -v llvm-profdata >/dev/null 2>&1 \
        || die "llvm-profdata не найден. Установите llvm (llvm-tools)."
    PGO_DATA="${PWD}/${PROF_DIR}/merged.profdata"
    llvm-profdata merge -o "$PGO_DATA" \
        "${PROF_DIR}"/*.profraw \
        || die "llvm-profdata merge не удался"
    PGO_USE_FLAGS=(-fprofile-instr-use="$PGO_DATA" -fprofile-correction)
    PGO_USE_LDFLAGS=(-fprofile-instr-use="$PGO_DATA")
else
    PGO_USE_FLAGS=(-fprofile-use="${PWD}/${PROF_DIR}"
                   -fprofile-correction -Wno-missing-profile)
    PGO_USE_LDFLAGS=(-fprofile-use="${PWD}/${PROF_DIR}")
fi

cmake -S . -B "$BUILD_DIR" -G "$GENERATOR" \
    "${COMMON_FLAGS[@]}" \
    -DSCALE_PGO=ON \
    -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF \
    -DCMAKE_CXX_FLAGS="${PGO_USE_FLAGS[*]}" \
    -DCMAKE_EXE_LINKER_FLAGS="${PGO_USE_LDFLAGS[*]}"

cmake --build "$BUILD_DIR" --parallel "$JOBS"

ok "PGO-сборка завершена"

print_artifacts "$BUILD_DIR" "$BUILD_TYPE" "Артефакты (PGO)"

# -----------------------------------------------------------------------------
# Опциональное сравнение с baseline (BENCH=1)
# -----------------------------------------------------------------------------
if [[ -n "${BENCH:-}" ]]; then
    log "Сравнение времени работы baseline vs PGO:"
    BASE_DIR="build-baseline"
    base_exe="${BASE_DIR}/${MAIN_EXE}"
    if [[ ! -x "$base_exe" ]]; then
        log "Baseline не найден — собираю ${BASE_DIR} (без PGO)"
        cmake -S . -B "$BASE_DIR" -G "$GENERATOR" \
            -DCMAKE_BUILD_TYPE="${BUILD_TYPE}" \
            -DCMAKE_CXX_COMPILER="${CXX_BIN}" \
            -DCMAKE_EXPORT_COMPILE_COMMANDS=ON \
            -DSCALE_PGO=OFF \
            -DSCALE_NATIVE="$([[ "$NATIVE" -eq 1 ]] && echo ON || echo OFF)" \
            -DSCALE_WERROR=OFF \
            -DSCALE_BUILD_TESTS=OFF
        cmake --build "$BASE_DIR" --parallel "$JOBS"
        base_exe="${BASE_DIR}/${MAIN_EXE}"
    fi
    pgo_exe="${BUILD_DIR}/${MAIN_EXE}"
    if [[ -x "$base_exe" && -x "$pgo_exe" ]]; then
        t1=$( { /usr/bin/time -f '%e' "$base_exe" "$SAMPLE_INPUT" >/dev/null; } 2>&1 || true )
        t2=$( { /usr/bin/time -f '%e' "$pgo_exe"  "$SAMPLE_INPUT" >/dev/null; } 2>&1 || true )
        printf '  %-10s baseline=%ss  pgo=%ss\n' "$MAIN_EXE" "$t1" "$t2"
    else
        warn "Не удалось получить пути к baseline/pgo бинарникам"
    fi
fi

exit 0

# =============================================================================
#  scale — единая утилита масштабирования MP2 / AC-3 / DTS Core (включая
#  автоматическое извлечение DTS Core из DTS-HD MA/HRA).
#
#  Кроссплатформенная сборка под Windows 10+ (MSVC, MinGW-w64 UCRT64)
#  и Linux 6+.
#
#  Требования:
#    - CMake >= 3.25 (для C++23 и современного target-API);
#    - GCC >= 13, Clang >= 17 или MSVC >= 19.35 (VS 2022 17.5+);
#    - Ninja (рекомендуется) или Makefiles / Visual Studio generator.
#
#  Ключевые возможности:
#    - C++23 (std::expected, std::span, std::jthread, std::format);
#    - единая статическая библиотека scale_common с общим движком;
#    - глубокая оптимизация: LTO, AVX2 baseline, /GL /LTCG, -fno-plt;
#    - опциональный PGO-режим через -DSCALE_PGO=ON;
#    - предупреждения уровня -Wall -Wextra -Wpedantic и /W4 /permissive-;
#    - опциональные unit-тесты через -DSCALE_BUILD_TESTS=ON;
#    - Windows-ресурсы: манифест, VERSIONINFO, иконка (resources/).
# =============================================================================

cmake_minimum_required(VERSION 3.25)

# -----------------------------------------------------------------------------
# libstdc++exp: начиная с GCC 14 реализация std::print / std::println
# вынесена в отдельную экспериментальную библиотеку libstdc++exp. При
# статической линковке на MinGW она не подтягивается автоматически, и
# линкер падает с "undefined reference to `std::__open_terminal'".
#
# MSVC использует собственную STL и этой библиотеки не требует. Clang с
# libc++ тоже не требует (и std::print там пока отсутствует — сработает
# fallback в include/scale/print.hpp). Поэтому линкуем только для GNU.
#
# Оформлено как INTERFACE-цель, чтобы автоматически подтягивалось всеми
# исполняемыми файлами проекта без дублирования в каждом target_link_libraries.
# -----------------------------------------------------------------------------

add_library(scale_print_backend INTERFACE)
# GNU и Clang (последний может использовать libstdc++ 14+).
# Если stdc++exp нет при GCC 14+, где __cpp_lib_print определён
# и std::print реально используется (fallback не срабатывает),
# линковка упадёт — это ошибка конфигурации, а не норма.
if((CMAKE_CXX_COMPILER_ID STREQUAL "GNU"
    AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 14)
   OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    find_library(SCALE_STDCXXEXP_LIB stdc++exp)
    if(SCALE_STDCXXEXP_LIB)
        target_link_libraries(scale_print_backend INTERFACE stdc++exp)
    endif()
endif()

# scale_print_backend подтянут на уровне scale_common: сама библиотека

# что оставляло scale_common без нужного символа при внешнем использовании.

# -----------------------------------------------------------------------------
# Компилятор-специфичные флаги
# -----------------------------------------------------------------------------

if(MSVC)

    target_compile_definitions(scale_common PUBLIC
        _WIN32_WINNT=0x0A00
        WINVER=0x0A00
        NOMINMAX
        WIN32_LEAN_AND_MEAN
        UNICODE
        _UNICODE
    )

    target_compile_options(scale_common PUBLIC
        /W4
        /permissive-
        /Zc:__cplusplus
        /Zc:preprocessor
        /Zc:inline
        /EHsc
        /utf-8
        /MP
    )

    target_compile_options(scale_common PUBLIC
        $<$<CONFIG:Release>:
            /O2 /Oi /Ot /Oy /Ob3 /GT /Gw /Gy /GL /fp:fast
        >
    )
    if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND
       CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        target_compile_options(scale_common PUBLIC
            $<$<CONFIG:Release>:/arch:AVX2>
        )
    endif()

    if(SCALE_WERROR)
        target_compile_options(scale_common PRIVATE /WX)
        target_compile_options(scale       PRIVATE /WX)
    endif()

    target_link_options(scale PRIVATE
        $<$<CONFIG:Release>:
            /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO
        >
    )

    set_property(TARGET scale PROPERTY MSVC_RUNTIME_LIBRARY
        "MultiThreaded$<$<CONFIG:Debug>:Debug>")

else()  # GCC / Clang (Linux, MinGW-w64 UCRT64, macOS)

    # --- общие предупреждения ------------------------------------------------
    target_compile_options(scale_common PUBLIC
        -Wall
        -Wextra
        -Wpedantic
        -Wshadow
        -Wconversion
        -Wsign-conversion
        -Wcast-qual
        -Wcast-align
        -Wdouble-promotion
        -Wformat=2
        -Wnon-virtual-dtor
        -Wold-style-cast
        -Woverloaded-virtual
        -Wnull-dereference
    )

    # --- предупреждения, поддерживаемые только GCC ---------------------------
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(scale_common PUBLIC
            -Wduplicated-cond
            -Wduplicated-branches
            -Wlogical-op
            -Wuseless-cast
        )
    endif()

    if(SCALE_WERROR)
        target_compile_options(scale_common PRIVATE -Werror)
        target_compile_options(scale       PRIVATE -Werror)
    endif()

    # --- выбор микроархитектуры ----------------------------------------------
    # SCALE_NATIVE=ON  → -march=native -mtune=native;
    # иначе            → x86-64-v3 = AVX2 + FMA + BMI1/2.
    # Baseline совместим со всеми Win 10-совместимыми CPU начиная
    # с Haswell (2013).
    if(SCALE_NATIVE)
        set(SCALE_ARCH_FLAGS -march=native -mtune=native)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        # Baseline AVX2: Intel Haswell (2013) / AMD Excavator (2015).
        set(SCALE_ARCH_FLAGS -march=x86-64-v3 -mtune=generic)
    else()
        # ARM64 и другие архитектуры: -march не задаём.
        set(SCALE_ARCH_FLAGS "")
    endif()

    # --- общие флаги Release для компиляции ----------------------------------
    # PUBLIC: флаги докатываются и до scale (main.cpp), который
    # линкуется с scale_common, иначе main.cpp собирался бы
    # без -O3 / -march.
    target_compile_options(scale_common PUBLIC
        $<$<CONFIG:Release>:
            -O3
            ${SCALE_ARCH_FLAGS}
            -fno-plt
            -fno-semantic-interposition
            -funroll-loops
            -fomit-frame-pointer
            -fstrict-aliasing
            -fmerge-all-constants
            -ffunction-sections
            -fdata-sections
        >
    )

    # --- флаги Release для линковки: общие для ELF и PE ---------------------
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(scale PRIVATE
            $<$<CONFIG:Release>:
                -Wl,-O1
                -Wl,--gc-sections
            >
        )
    elseif(APPLE)
        target_link_options(scale PRIVATE
            $<$<CONFIG:Release>:-Wl,-dead_strip>
        )
    endif()

    # --- флаги Release для линковки: только ELF -----------------------------
    # -fuse-linker-plugin, -Wl,--as-needed, -Wl,-z,now и -Wl,-z,relro
    # понимает только GNU ld для ELF. PE-линкер x86_64-w64-mingw32-ld
    # в MSYS2 падает на "unrecognized option '-z'". Поэтому на Windows
    # эти флаги пропускаются.
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(scale PRIVATE
            $<$<CONFIG:Release>:
                -Wl,--as-needed
                -Wl,-z,now
                -Wl,-z,relro
            >
        )
        # -fuse-linker-plugin — опция драйвера GCC; Clang её
        # не понимает и падает с \"unknown argument\".
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            target_link_options(scale PRIVATE
                $<$<CONFIG:Release>:-fuse-linker-plugin>
            )
        endif()
    endif()

endif()

# =============================================================================
# Windows-ресурсы: манифест, VERSIONINFO, иконка.
#
# Все три файла лежат в resources/:
#   * scale.rc.in    — шаблон ресурсного файла; CMake подставит версию;
#   * scale.manifest.in — шаблон XML-манифеста; CMake подставит
#                          актуальную версию (DPI, longPathAware,
#                          supportedOS, requestedExecutionLevel);
#   * icon.ico       — иконка приложения (встраивается как RT_GROUP_ICON).
#
# configure_file(@ONLY) генерирует scale.rc в бинарном каталоге с
# абсолютными путями к манифесту и иконке. Forward slashes в путях
# понимают и rc.exe (MSVC), и windres (MinGW). Сгенерированный .rc
# подключается к цели scale; CMake сам вызовет нужный ресурсный
# компилятор и слинкует .res.
#
# Манифест встраивается через .rc (ресурс RT_MANIFEST, ID 1). На MSVC
# это требует отключить автоматическую генерацию манифеста линкером
# (/MANIFEST:NO), иначе возможен конфликт двух RT_MANIFEST.
# =============================================================================

if(WIN32)
    if(MINGW)
        # Отключаем автоматическую генерацию манифеста линкером
        # (CMake 3.30+ игнорирует .manifest для MinGW, вызывая конфликт)
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--disable-auto-import")
        # Флаги для windres: UTF-8 и обход бага с popen
        set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} --use-temp-file -c 65001")
    endif()
    set(SCALE_RESOURCES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/resources")

    # Проверяем наличие обязательных файлов. Без этой проверки
    # configure_file() сгенерирует .rc с битыми путями, и сборка
    # упадёт с неинформативной ошибкой линкера.
    foreach(_res_file
            "${SCALE_RESOURCES_DIR}/scale.rc.in"
            "${SCALE_RESOURCES_DIR}/scale.manifest.in"
            "${SCALE_RESOURCES_DIR}/icon.ico")
        if(NOT EXISTS "${_res_file}")
            message(FATAL_ERROR
                "Windows-ресурс не найден: ${_res_file}\n"
                "Ожидаются: resources/scale.rc.in, resources/scale.manifest.in, "
                "resources/icon.ico.")
        endif()
    endforeach()

    # @ONLY: подставляются только @VAR@, последовательности ${VAR}
    # остаются нетронутыми. В .rc могут встречаться фрагменты,
    # похожие на ${...}, которые не должны интерпретироваться CMake.
    configure_file(
        "${SCALE_RESOURCES_DIR}/scale.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/scale.rc"
        @ONLY
    )




    # Манифест тоже генерируется из шаблона: версия синхронизирована
    # с project(... VERSION ...). Это устраняет рассинхронизацию
    # между common.hpp / CMakeLists.txt / манифестом.
    configure_file(
        "${SCALE_RESOURCES_DIR}/scale.manifest.in"
        "${CMAKE_CURRENT_BINARY_DIR}/scale.manifest"
        @ONLY
    )

    # Подключаем сгенерированный .rc к цели scale.
    target_sources(scale PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/scale.rc")

    # MSVC: отключаем автогенерацию манифеста линкером, чтобы не
    # было двух RT_MANIFEST (наш из .rc + автоматический).
    if(MSVC)
        target_link_options(scale PRIVATE "/MANIFEST:NO")
    endif()
endif()

# Русский

## Зачем это нужно

Когда записанные цифровые телепрограммы перекодируются для DVD или VCD, часто видео и аудио расходятся по времени. Если сдвиг фиксированный — справится любая программа резки MPEG. Но бывает, что рассинхронизация накапливается постепенно: один поток оказывается длиннее другого на доли секунды. Этот проект решает именно такую задачу — **немного растянуть или сжать аудиопоток**, чтобы его длина совпала с видеорядом.

## Как это работает

Аудиопотоки MP2, AC-3 и DTS Core состоят из **самодостаточных кадров** сжатых сэмплов: ни один кадр не ссылается на данные предыдущих. Поэтому:

- если поток нужно **растянуть** — утилита равномерно вставляет дублированные кадры;
- если **сжать** — равномерно удаляет отдельные кадры.

Подход грубоват, зато очень быстр, не меняет высоту тона и не требует ресемплинга. Для подгонки A/V-синхронизации этого более чем достаточно: точность — в пределах одного аудиокадра (~20–35 мс: MP2 ≈ 24 мс, AC-3 ≈ 32 мс при 48 кГц, DTS — по NBLKS).

> **Почему это не работает для MP3 и AAC?** Эти форматы используют битовый резервуар: кадр может заимствовать биты у предыдущих. Наивное дублирование или удаление кадра сломало бы поток. Поэтому MP3 и AAC не поддерживаются.

## Использование

```
scale [-t target] [-o outfile] infile
scale --help
```

| Режим | Что делает |
|---|---|
| Без `-t` и без `-o` | Только отчёт: тип аудио, каналы, размер кадра, частота, битрейт, длительность, число кадров. |
| С `-t`, без `-o` | «Сухой прогон»: сообщает, сколько кадров было бы вставлено или удалено. |
| С `-t` и `-o` | Масштабирует поток до указанной длины и записывает результат в `outfile`. |

### Примеры

```bash
# Растянуть audio.mp2 до 20.5 секунд
scale -t 20.5 -o newaudio.mp2 audio.mp2

# Ровно 500 кадров
scale -t 500f -o newaudio.ac3 audio.ac3

# До отметки 00:20:30 (hh:mm:ss)
scale -t 00:20:30 -o newaudio.dts audio.dts

# Только отчёт о свойствах входного файла
scale audio.mp2

# Сухой прогон: что было бы сделано
scale -t 20.5 audio.mp2
```

## Опция `-t`

Целевую длительность можно задать в одной из четырёх форм:

| Форма | Пример | Значение |
|---|---|---|
| Секунды (по умолчанию) | `20.5` | 20.5 секунд |
| Секунды (явно) | `20.5s` | 20.5 секунд |
| Кадры | `500f` | 500 кадров |
| Время | `01:30` | 1 мин 30 сек |
| Время | `00:20:30` | 20 мин 30 сек |

Допускается дробная точность в форме `00:20:30.500`; для кадров — только целое число. Для формы «кадры» движок использует среднюю длительность кадра по всему потоку, поэтому результат точен даже для потоков с переменным размером кадра (MP2 padding, DTS с меняющимся NBLKS).

## Поддерживаемые форматы

| Формат | Синхрослово | Особенности |
|---|---|---|
| **MP2** (MPEG-1 Layer II) | `0xFFF` (11 бит) | Размер/длительность из индексов битрейта и частоты. Padding учитывается покадрово. Layer I и III (MP3) отклоняются. |
| **AC-3** (Dolby Digital) | `0x0B77` | Размер кадра — по таблице ATSC A/52 для 48/44.1/32 кГц. Кадры фиксированные. |
| **DTS Core** | `0x7FFE8001` (BE) / `0xFE7F0180` (LE) | NBLKS × 32 сэмпла, FSIZE + 1 байт. Таблицы из ETSI TS 102 114. |
| **DTS-HD MA/HRA** | — | **Извлекается ядро** DTS Core, XLL/X96-расширение отбрасывается. На выходе — lossy DTS Core (битрейт берётся из заголовка ядра); воспроизводится любым DTS-декодером. |

Все форматы определяются автоматически: сначала DTS, затем AC-3, затем MP2. Такой порядок гарантирует, что длинное синхрослово DTS не будет ошибочно принято за мусор.

```
scale/
├── include/scale/
│   ├── common.hpp          Публичный API: Codec, Args, TargetSpec, detect_codec()
│   └── print.hpp           Shim для std::print (GCC 13 / Clang 17)
├── src/
│   ├── common.cpp          Движок: detection, поиск кадров, конвейер
│   ├── mp2codec.cpp        MP2: парсинг заголовка, padding
│   ├── ac3codec.cpp        AC-3: таблица ATSC A/52 на три частоты
│   ├── dtscodec.cpp        DTS Core + извлечение ядра из DTS-HD
│   └── main.cpp            Единственная точка входа
├── tests/                  Unit- и интеграционные тесты
├── build.sh                Кроссплатформенная сборка (Linux/macOS)

├── build_msys2.sh          Статическая сборка в MSYS2 UCRT64
└── CMakeLists.txt
```

**Ключевые идеи:**

- **Один бинарник.** Все три формата обрабатываются одним `scale`; выбор кодека — через `Codec*`-фабрику.
- **Покадровый парсинг.** Движок вызывает `parse_header()` на каждом кадре, поэтому корректно работает с MP2 padding и DTS-HD extension.
- **`frame_span()`** — точка расширения: базовая реализация возвращает `fi.framebytes`, DTS переопределяет и сканирует следующий sync, отбрасывая XLL/X96.
- **Многопоточность.** Конвейер reader/writer (`std::jthread` + `BoundedQueue`), перекрывающий чтение, обработку и запись.
- **Bounds-safe.** Везде `std::span`, все проверки границ — до чтения. Никаких `strcpy` и `argv[++i]` без проверки.

### По сравнению с оригиналом 2004 года

- Исправлено чтение за границей буфера в цикле поиска кадров.
- Устранено беззнаковое переполнение в расчёте хвостового мусора.
- Убран `fseek(-10000)`, ломавшийся на коротких файлах.
- Исправлен разбор channel mode в MP2 (`&&` → `&`).
- Исправлены формулы размера кадра AC-3 для 44.1 и 32 кГц.
- Все `%d` заменены на `std::format`.

# English

## Motivation

When captured digital TV programs are reencoded for DVD or VCD, video and audio often drift out of sync. A fixed offset is easy to fix with any MPEG cutting tool. But sometimes the drift accumulates gradually: one stream ends up slightly longer than the other. This project solves exactly that — **slightly stretch or shrink the audio stream** so its length matches the video.

## How it works

MP2, AC-3 and DTS Core streams consist of **self-contained frames**: no frame refers back to data in previous frames. Therefore:

- to **stretch** a stream, the tool inserts evenly-spread duplicated frames;
- to **shrink** it, it drops a frame once in a while.

The approach is crude but very fast, keeps the original pitch, and requires no resampling. For A/V sync correction this is more than enough: accuracy is within one audio frame (~20–35 ms: MP2 ≈ 24 ms, AC-3 ≈ 32 ms @ 48 kHz, DTS — depends on NBLKS).

> **Why doesn't this work for MP3 and AAC?** Those formats use a bit reservoir: a frame can borrow bits from previous frames. Naively duplicating or dropping a frame would break the stream. Hence MP3 and AAC are not supported.

## Usage

```
scale [-t target] [-o outfile] infile
scale --help
```

| Mode | What it does |
|---|---|
| Without `-t` and `-o` | Report only: audio type, channels, frame size, sample rate, bitrate, duration, frame count. |
| With `-t`, without `-o` | Dry run: reports how many frames would be inserted or dropped. |
| With `-t` and `-o` | Scales the stream to the target length and writes it to `outfile`. |

### Examples

```bash
# Stretch audio.mp2 to 20.5 seconds
scale -t 20.5 -o newaudio.mp2 audio.mp2

# Exactly 500 frames
scale -t 500f -o newaudio.ac3 audio.ac3

# Up to 00:20:30 (hh:mm:ss)
scale -t 00:20:30 -o newaudio.dts audio.dts

# Report properties only
scale audio.mp2

# Dry run: what would be done
scale -t 20.5 audio.mp2
```

## The `-t` option

The target length can be given in one of four forms:

| Form | Example | Meaning |
|---|---|---|
| Seconds (default) | `20.5` | 20.5 seconds |
| Seconds (explicit) | `20.5s` | 20.5 seconds |
| Frames | `500f` | 500 frames |
| Time | `01:30` | 1 min 30 sec |
| Time | `00:20:30` | 20 min 30 sec |

Fractional precision is accepted in time form (`00:20:30.500`); frames must be integers. For the frame form the engine uses the average frame duration across the whole stream, so the result is exact even for streams with variable frame size (MP2 padding, DTS with changing NBLKS).

## Supported formats

| Format | Sync word | Notes |
|---|---|---|
| **MP2** (MPEG-1 Layer II) | `0xFFF` (11 bits) | Frame size/duration from bitrate and frequency indices. Padding handled per frame. Layer I and III (MP3) are rejected. |
| **AC-3** (Dolby Digital) | `0x0B77` | Frame size from the ATSC A/52 table for 48/44.1/32 kHz. Fixed-size frames. |
| **DTS Core** | `0x7FFE8001` (BE) / `0xFE7F0180` (LE) | NBLKS × 32 samples, FSIZE + 1 bytes. Tables from ETSI TS 102 114. |
| **DTS-HD MA/HRA** | — | **DTS Core is extracted**; the XLL/X96 extension is discarded. Output is a lossy DTS Core stream (bitrate taken from the Core header) playable by any DTS decoder. |

All formats are auto-detected: DTS first, then AC-3, then MP2. This order ensures the long DTS sync word is never mistaken for junk.

## Architecture

```
scale/
├── include/scale/
│   ├── common.hpp          Public API: Codec, Args, TargetSpec, detect_codec()
│   └── print.hpp           std::print shim (GCC 13 / Clang 17)
├── src/
│   ├── common.cpp          Engine: detection, frame scanning, pipeline
│   ├── mp2codec.cpp        MP2: header parsing, padding
│   ├── ac3codec.cpp        AC-3: ATSC A/52 table for three sample rates
│   ├── dtscodec.cpp        DTS Core + DTS-HD core extraction
│   └── main.cpp            Single entry point
├── tests/                  Unit and integration tests
├── build.sh                Cross-platform build (Linux/macOS)
├── build.ps1               Windows build (MSVC/clang-cl)
├── build_msys2.sh          Static build in MSYS2 UCRT64
└── CMakeLists.txt
```

**Key ideas:**

- **One binary.** All three formats are handled by a single `scale`; codec selection goes through a `Codec*` factory.
- **Per-frame parsing.** The engine calls `parse_header()` on every frame, so MP2 padding and DTS-HD extension are handled correctly.
- **`frame_span()`** is the extension point: the base implementation returns `fi.framebytes`; DTS overrides it and scans for the next sync, discarding the XLL/X96 extension.
- **Concurrency.** A reader/writer pipeline (`std::jthread` + `BoundedQueue`) that overlaps reading, processing and writing.
- **Bounds-safe.** Everything goes through `std::span`; all size checks happen before access. No `strcpy`, no unchecked `argv[++i]`.

### Compared to the original 2004 version

- Fixed out-of-bounds read in the frame-search loop.
- Removed unsigned underflow in the trailing-junk calculation.
- Removed `fseek(-10000)`, which broke on short files.
- Fixed MP2 channel-mode parse (`&&` → `&`).
- Fixed AC-3 frame-size formulas for 44.1 and 32 kHz.
- Replaced all `%d` with `std::format`.
